% Beamer style modeled on the Applied Crypto Group / ETH lecture style.

\usepackage[T1]{fontenc}
\usepackage[defaultsans]{lato}
\renewcommand*\familydefault{\sfdefault}

% Warm blue accent (steel-blue family, slightly warm).
\definecolor{acgred}{RGB}{56,103,170}
\definecolor{bodygray}{RGB}{77,77,77}
\definecolor{blockgray}{RGB}{236,236,236}
\definecolor{blocktitlegray}{RGB}{215,215,215}

% Highlight palette — soft pastels for inline term highlighting.
\definecolor{hlyellow}{RGB}{255,239,170}
\definecolor{hlpeach}{RGB}{255,214,196}

\definecolor{hlblue}{RGB}{200,226,244}
\definecolor{hlpurple}{RGB}{223,213,243}
\definecolor{hlred}{RGB}{255,200,200}

% Highlight macro: works in both text and math mode.
% Usage: \hly{text}, \hlp{text}, \hlg{text}, \hlb{text}, \hlv{text}, \hlr{text}.
\newcommand{\acghl}[2]{%
  \tikz[baseline=(X.base)]\node[%
    inner xsep=2pt, inner ysep=1pt, outer sep=0pt,%
    rectangle, fill=#1, rounded corners=1pt%
  ] (X) {#2};%
}
\newcommand{\hly}[1]{\acghl{hlyellow}{#1}}
\newcommand{\hlp}[1]{\acghl{hlpeach}{#1}}
\newcommand{\hlg}[1]{\acghl{hlgreen}{#1}}
\newcommand{\hlb}[1]{\acghl{hlblue}{#1}}
\newcommand{\hlv}[1]{\acghl{hlpurple}{#1}}
\newcommand{\hlr}[1]{\acghl{hlred}{#1}}

% Single source of truth for horizontal margins on all slides.
\newcommand{\acgsidemargin}{0.06\paperwidth}

\setbeamerfont{title}{size=\Huge, series=\mdseries}
\setbeamerfont{subtitle}{size=\Huge, series=\mdseries}
\setbeamerfont{author}{size=\normalsize, series=\mdseries}
\setbeamerfont{institute}{size=\normalsize, series=\mdseries}
\setbeamerfont{frametitle}{size=\Large, series=\mdseries}
\setbeamerfont{framesubtitle}{size=\large, series=\mdseries}
\setbeamerfont{block title}{size=\small, series=\mdseries}
\setbeamerfont{block body}{size=\small, series=\mdseries}
\setbeamerfont{itemize/enumerate body}{size*={8.5pt}{10pt}}
\setbeamerfont{itemize/enumerate subbody}{size*={8.5pt}{10pt}}
\setbeamerfont{itemize/enumerate subsubbody}{size*={8.5pt}{10pt}}
\setbeamerfont{page number in head/foot}{size=\small}

\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{itemize item}{\textbullet}
\setbeamertemplate{itemize subitem}{\textbullet}
\setbeamertemplate{itemize subsubitem}{\textbullet}

\setbeamertemplate{frametitle}{%
  \nointerlineskip
  \vspace{1.8em}
  \begin{beamercolorbox}[wd=\paperwidth, leftskip=\acgsidemargin, rightskip=\acgsidemargin]{frametitle}%
    \usebeamerfont{frametitle}\insertframetitle\par%
    \ifx\insertframesubtitle\empty\else
      \vspace{0.15em}
      {\usebeamerfont{framesubtitle}\usebeamercolor[fg]{framesubtitle}\insertframesubtitle\par}%
    \fi
  \end{beamercolorbox}%
  \vspace{0.5em}
}

\setbeamertemplate{footline}{%
  \begin{beamercolorbox}[wd=\paperwidth, ht=3.2ex, dp=2.0ex, leftskip=\acgsidemargin, rightskip=0.015\paperwidth]{}%
    \hfill
    {\usebeamerfont{page number in head/foot}%
     \usebeamercolor[fg]{page number in head/foot}%
     \insertframenumber}%
  \end{beamercolorbox}%
}

\setbeamertemplate{title page}{%
  \vbox to \dimexpr\paperheight-2em\relax{%
    \vfil
    \begingroup
      \centering
      {\usebeamerfont{title}\usebeamercolor[fg]{title}\inserttitle\par}%
      \ifx\insertsubtitle\empty\else
        \vspace{0.55em}
        {\usebeamerfont{subtitle}\usebeamercolor[fg]{subtitle}\insertsubtitle\par}%
      \fi
      \vspace{2.4em}
      {\usebeamerfont{author}\usebeamercolor[fg]{author}\insertauthor\par}%
      \ifx\insertinstitute\empty\else
        \vspace{0.45em}
        {\usebeamerfont{institute}\usebeamercolor[fg]{institute}\insertinstitute\par}%
      \fi
      \ifx\insertdate\empty\else
        \vspace{0.45em}
        {\usebeamerfont{date}\usebeamercolor[fg]{date}\insertdate\par}%
      \fi
    \endgroup
    \vfil
  }%
}
