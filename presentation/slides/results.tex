\section{Results}

\begin{frame}
    \frametitle{Robust Black-Box DAG Combiners}
    \textbf{Idea.}
    Represent each algorithm of the combiner as a \emph{directed acyclic graph} (DAG) $\G$
    that makes the structure of oracle calls explicit.

    \vspace{0.5em}
    \begin{itemize}
        \item Dedicated \textbf{source} $s$ (the input) and \textbf{sink} $t$ (the output).
        \item Every other vertex $v$ carries
            \begin{itemize}
                \item a label $\ell(v) = (i,j)$ specifying \emph{which candidate's} $j$-th oracle is queried, and
                \item a ppt.\ Turing machine $f^v$ that computes the query input from the responses
                    of the predecessors $\mathcal N^-_\G(v)$.
            \end{itemize}
        \item Execution traces the DAG in topological order:
            at each $v$, build $\mathbf q_v \gets f^v(\{\mathbf x_w\}_{w \in \mathcal N^-_\G(v)})$
            and set $\mathbf x_v \gets \mathcal O_{\ell(v)}(\mathbf q_v)$.
            The sink's Turing machine $f^t$ produces the final output.
    \end{itemize}

    \vspace{0.3em}
    \textbf{Equivalence.}
    Every robust $(n;k)$ black-box combiner has a \emph{computationally equivalent}
    DAG combiner --- the DAG form is a convenient normal form for our proofs.
\end{frame}

\begin{frame}
    \frametitle{Example: PRF-Combiner as a DAG}
    The standard $(2;1)$ black-box PRF-combiner
    $\alg{Eval}{PRF}^{\mathcal O_1, \mathcal O_2}(k, x) := \mathcal O_1(k,x) \oplus \mathcal O_2(k,x)$
    as a DAG:

    \vspace{0.5em}
    \begin{center}
        \begin{tikzpicture}[
            every node/.style={font=\small},
            vertex/.style={draw, rounded corners, inner sep=4pt, align=center, minimum width=2.4cm},
            arr/.style={-{Latex[length=2mm]}}
        ]
            \node[vertex] (s) at (0, 2)
                {\textbf{Source $s$}\\ $\mathbf x_s \gets (k, x)$};
            \node[vertex] (v1) at (-3, 0)
                {\textbf{$v_1$}\\ $\mathbf q_{v_1} \gets \mathbf x_s$\\
                $\mathbf x_{v_1} \gets \mathcal O_1(\mathbf q_{v_1})$};
            \node[vertex] (v2) at (3, 0)
                {\textbf{$v_2$}\\ $\mathbf q_{v_2} \gets \mathbf x_s$\\
                $\mathbf x_{v_2} \gets \mathcal O_2(\mathbf q_{v_2})$};
            \node[vertex] (t) at (0, -2)
                {\textbf{Sink $t$}\\ $\mathbf y \gets \mathbf x_{v_1} \oplus \mathbf x_{v_2}$};
            \draw[arr] (s) -- (v1);
            \draw[arr] (s) -- (v2);
            \draw[arr] (v1) -- (t);
            \draw[arr] (v2) -- (t);
        \end{tikzpicture}
    \end{center}
\end{frame}

\begin{frame}
    \frametitle{Example: reading the DAG}
    \begin{itemize}
        \item The \textbf{source} packages the algorithm's inputs $(k, x)$ into $\mathbf x_s$.
        \item Vertices $v_1, v_2$ are labeled with $\ell(v_i) = (i, \alg{Eval}{PRF})$;
            each queries its candidate with the forwarded input and stores the response $\mathbf x_{v_i}$.
        \item The \textbf{sink}'s Turing machine combines the predecessors' outputs,
            returning $\mathbf x_{v_1} \oplus \mathbf x_{v_2}$.
        \item The DAG makes explicit what the usual pseudocode hides:
            \begin{itemize}
                \item which queries happen in parallel vs.\ in sequence,
                \item which candidate each query targets,
                \item exactly what information each query depends on.
            \end{itemize}
        \item This structure is what lets us argue about the combiner's computation
            vertex-by-vertex in the impossibility proof.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{What We Showed}
    \begin{block}{DAG combiners are a normal form}
        Every robust $(n;k)$ black-box combiner has a computationally equivalent
        robust $(n;k)$ black-box DAG combiner --- we can reason over DAGs without loss.
    \end{block}

    \begin{block}{Impossibility}
        There is \textbf{no} robust $(2;1)$ black-box VRF-combiner.\\
        By reduction, no robust $(n; \lfloor n/2 \rfloor)$ black-box VRF-combiner exists either ---
        at least half of the candidates must be secure.
    \end{block}

    \begin{block}{Matching positive result}
        If we \emph{strengthen} the candidates by assuming every one is uniquely provable and
        perfectly correct, then a simple XOR-style robust $(n;1)$ black-box VRF-combiner exists.
    \end{block}

    \vspace{0.2em}
    \textbf{Takeaway.}
    Combining VRFs is strictly harder than combining PRFs:
    verifiability forces the combiner to commit to a single provable image,
    and without strong guarantees on the candidates, no black-box combiner can do this.
\end{frame}

\begin{frame}
    \frametitle{Beyond Impossibility}
    The impossibility result rules out black-box VRF-combiners in the strictest setting.
    Several natural relaxations are worth exploring:

    \vspace{0.5em}
    \begin{itemize}
        \item \textbf{Statistical correctness.}
            Require the combined scheme to be correct only with overwhelming probability,
            rather than perfectly --- does this open up new constructions?
        \item \textbf{Stronger candidates.}
            Our positive result assumes \emph{all} candidates are uniquely provable and perfectly correct.
            How far can we weaken this (e.g.\ statistical uniqueness, a mix of assumptions) and still combine?
        \item \textbf{Richer VRF notions.}
            Combiners for VRF variants with additional algebraic structure:
            \begin{itemize}
                \item constrainable VRFs,
                \item key-homomorphic VRFs,
                \item aggregable VRFs,
                \item unbiasable VRFs.
            \end{itemize}
            These primitives may admit combiners even when plain VRFs do not.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Open Questions}
    \begin{block}{Conjecture: $(n; n-1)$ is also impossible}
        There exists no robust $(n; n-1)$ black-box VRF-combiner.\\[0.3em]
        That is, it is \emph{not} enough for all-but-one candidates to be secure ---
        any robust black-box VRF-combiner needs strictly more than a constant
        number of insecure candidates to tolerate.
    \end{block}

    \vspace{0.3em}
    \textbf{Why it is hard.}
    Our $(2;1)$ impossibility proof exploits a specific structural argument
    about obscure queries and verification traces.
    Extending it to $n-1$ adversarial candidates would require reasoning
    about the simultaneous freedom on many oracles, and the current technique
    does not directly generalize.

    \vspace{0.3em}
    \textbf{Other open directions.}
    \begin{itemize}
        \item Closing the gap between $(n; \lfloor n/2 \rfloor)$ (impossible)
            and $(n; n-1)$ (conjectured impossible).
        \item Finding the weakest candidate assumptions under which
            \emph{some} robust black-box VRF-combiner exists.
    \end{itemize}
\end{frame}
