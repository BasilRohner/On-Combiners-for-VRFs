\begin{frame}
    \frametitle{Robust Black-Box DAG Combiners}

    We consider robust black-box combiners in a DAG normal form.

    \begin{itemize}
        \item Dedicated \textbf{source} $s$ (the input) and \textbf{sink} $t$ (the output).
        \item Every other vertex $v$ carries
            \begin{itemize}
                \item a label $\ell(v) = (i,j)$ specifying \emph{which candidate's} $j$-th oracle is queried, and
                \item a ppt.\ Turing machine $f^v$ that computes the query input from the responses
                    of the predecessors $\mathcal N^-_\G(v)$.
            \end{itemize}
        \item Execution traces the DAG in topological order:
            at each $v$, build $\mathbf q_v \gets f^v(\{\mathbf x_w\}_{w \in \mathcal N^-_\G(v)})$
            and set $\mathbf x_v \gets \mathcal O_{\ell(v)}(\mathbf q_v)$.
            The sink's Turing machine $f^t$ produces the final output.
    \end{itemize}

    \vspace{0.3em}
    \textbf{Equivalence.}
    Every robust $(n;k)$ black-box combiner has a \emph{computationally equivalent}
    DAG combiner --- the DAG form is a convenient normal form for our proofs.
\end{frame}

\begin{frame}
    \frametitle{Proof Setup}

    Suppose $\BBComb{VRF}$ is a robust $(2;1)$ black-box DAG VRF-combiner.

    \begin{itemize}
        \item $\mathcal O_{vrf}$: a secure VRF
        \item $\mathcal O_{uppc}$: a \emph{trivial}, uniquely provable and perfectly correct VF:
            \begin{itemize}
                \item $\alg{Gen}{VRF}(1^\lambda) = (\sk_{def}, \pk_{def})$
                \item $\alg{Eval}{VRF}(\sk, x) = (y_{def}, \pi_{def})$
                \item $\alg{Vfy}{VRF}(\pk, x, y, \pi) = 1 \iff y = y_{def}$
            \end{itemize}
        \item $\mathcal O_{vrf}$ a secure VRF $\implies$ $\BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]$ a secure VRF
    \end{itemize}

    \vspace{0.3em}

    \textbf{Goal:} Adversary $\mathcal A^{\mathsf{PSPACE}, \mathcal O_{vrf}, \mathcal O_{uppc}}$
    against psuedorandomness of $\BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]$.

    \textbf{Assume}: $\SK = \bin^{s(\lambda)}$ for $s \in \poly(\lambda)$ for all candidates.

\begin{frame}
    \frametitle{Proof Overview - How to get $\alg{EvalSim}{VRF}$}

    Given $(\sk_{cand}, x, y, \pi, \tau)$, $\alg{EvalSim}{VRF}$ is computed by:

    \begin{itemize}
        \item Follow the DAG $\G_{\text{Eval}}$, describing $\BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]$, in some topological ordering
        \item For every vertex $v$ corresponding to a query to $\mathcal O_{uppc}$ answer with default value
        \item For every vertex $v$ corresponding to a query to $\mathcal O_{vrf}$ check $\tau$:
        \begin{itemize}
            \item If $\mathbf q \in \tau$: answer accordingly
            \item If $\mathbf q \notin \tau$: branch for every possible answer
        \end{itemize}
        \item Return $\top$ iff all branches produce the image $y$
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Exposed vs.\ Obscure Queries}

    Honest evaluation $(y,\pi) \gets \BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}].\alg{Eval}{VRF}(\sk, x)$;
    let $\tau$ be the verification trace. Classify each oracle-query vertex of the Eval DAG:

    \begin{itemize}
        \item query to $\mathcal O_{uppc}$ \hfill$\to$\quad default $\mathbf x_v^{def}$
        \item query to $\mathcal O_{vrf}$ in $\tau$ \hfill$\to$\quad \emph{exposed}: read from $\tau$
        \item query to $\mathcal O_{vrf}$ not in $\tau$ \hfill$\to$\quad \emph{obscure}
    \end{itemize}

    \begin{block}{Claim (obscure queries are irrelevant)}
        Under any well-formed reassignment of responses to the obscure queries,
        the combiner still outputs the same image $y$.
    \end{block}

    \textbf{Sketch.} Else two reassignments yield $\widehat{\mathcal O}_{vrf}, \widetilde{\mathcal O}_{vrf}$
    with $\hat y \neq \tilde y$. The \emph{vrfification lemma} upgrades $\mathcal O_{uppc}$ to a secure VRF
    matching it on the relevant keys --- $\tau$ is the same in both worlds, breaking unique provability. \contradiction
\end{frame}

\begin{frame}
    \frametitle{A Query-Free Simulator $\alg{EvalSim}{VRF}$}

    Given $(\sk, x, y, \pi, \tau)$, simulate the combiner's Eval without any oracle call:
    answer $\mathcal O_{uppc}$ queries with defaults, exposed queries from $\tau$, and \emph{branch}
    over all well-formed responses on obscure queries. Return $\top$ iff every branch outputs $y$.

    \begin{block}{Properties}
        \begin{itemize}
            \item $\alg{EvalSim}{VRF}$ makes no queries to $\mathcal O_{vrf}, \mathcal O_{uppc}$.
            \item $\sk = \sk^* \Rightarrow \alg{EvalSim}{VRF}(\sk, x, y, \pi, \tau) = \top$ \hfill(previous claim)
            \item $\BBComb{VRF}.\alg{Eval}{VRF}^y(\sk, x) \neq y \Rightarrow \alg{EvalSim}{VRF}(\sk, x, y, \pi, \tau) = \bot$
        \end{itemize}
    \end{block}

    Inefficient, but decidable in $\mathsf{PSPACE}$.
\end{frame}

\begin{frame}
    \frametitle{The Adversary}

    \begin{itemize}
        \item[\textbf{1.}] Sample $x_1, \ldots, x_{s(\lambda) + \lambda} \sample \bin^\lambda$;
            obtain $(y_i, \pi_i, \tau_i)$ via $\mathcal O_{\text{eval}}$ and $\BBComb{VRF}.\alg{Vfy}{VRF}$.
        \item[\textbf{2.}] Use $\mathsf{PSPACE}$ to compute the filtered set
            $\SK_{cand} = \bigcap_{i} \{\sk \in \SK : \alg{EvalSim}{VRF}(\sk, x_i, y_i, \pi_i, \tau_i) = \top\}$;
            note $\sk^* \in \SK_{cand}$.
        \item[\textbf{3.}] Sample $\sk_{cand} \sample \SK_{cand}$ uniformly.
        \item[\textbf{4.}] On challenge $x^*$, predict $y' \gets \BBComb{VRF}.\alg{Eval}{VRF}(\sk_{cand}, x^*)$
            and guess $b = 0 \iff y' = y^*$.
    \end{itemize}

    \textbf{Advantage.} Call $\sk$ \emph{unfavorable} if it agrees with $\sk^*$ on a random $x$ with probability $< 1/2$.
    A Markov bound shows unfavorables survive all rounds with expected count $\le 2^{-\lambda}$, hence
    \[
        \mathbf{Adv}^{\textup{prf}}_{\BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]}(\lambda, \mathcal A)
        \ge \tfrac{1}{2} - \tfrac{1}{2^{\lambda + 1}} - \tfrac{1}{|\Y|}.
    \]
\end{frame}

\begin{frame}
    \frametitle{From $(2;1)$ to $(n;\lfloor n/2\rfloor)$}

    Given a robust $(n;\lfloor n/2\rfloor)$ black-box VRF-combiner $\BBComb{VRF}$, build a robust $(2;1)$ combiner $\DAGComb{VRF}$:

    \begin{itemize}
        \item initialize the first $\lfloor n/2\rfloor$ candidates of $\BBComb{VRF}$ with $\mathcal O_1$
        \item initialize the remaining $\lceil n/2\rceil$ candidates with $\mathcal O_2$
    \end{itemize}

    If either $\mathcal O_1$ or $\mathcal O_2$ is a secure VRF,
    at least $\lfloor n/2\rfloor$ candidates of $\BBComb{VRF}$ are secure, so $\DAGComb{VRF}$ is a robust $(2;1)$ combiner.

    \vspace{0.3em}
    Contraposition with the main theorem yields the impossibility of robust $(n;\lfloor n/2\rfloor)$ black-box VRF-combiners.
\end{frame}

\begin{frame}
    \frametitle{What We Showed}
    \begin{block}{Theorem. Impossibility $1$-out-of-$2$ combiner.}
        There is \textbf{no} robust $(2;1)$ black-box VRF-combiner.\\
    \end{block}
    \begin{block}{Corollary. Impossibility $\lfloor n/2\rfloor$-out-of-$n$ combiner.}
        There is \textbf{no} robust $(n;\lfloor n/2\rfloor)$ black-box VRF-combiner.\\
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Open Questions}
    \begin{block}{Conjecture: $(n; n-1)$ is also impossible}
        There exists no robust $(n; n-1)$ black-box VRF-combiner.\\[0.3em]
        That is, it is \emph{not} enough for all-but-one candidates to be secure ---
        any robust black-box VRF-combiner needs strictly more than a constant
        number of insecure candidates to tolerate.
    \end{block}

    \vspace{0.3em}
    \textbf{Why it is hard.}
    Our $(2;1)$ impossibility proof exploits a specific structural argument
    about obscure queries and verification traces.
    Extending it to $n-1$ adversarial candidates would require reasoning
    about the simultaneous freedom on many oracles, and the current technique
    does not directly generalize.

    \vspace{0.3em}
    \textbf{Other open directions.}
    \begin{itemize}
        \item Closing the gap between $(n; \lfloor n/2 \rfloor)$ (impossible)
            and $(n; n-1)$ (conjectured impossible).
        \item Finding the weakest candidate assumptions under which
            \emph{some} robust black-box VRF-combiner exists.
    \end{itemize}
\end{frame}
