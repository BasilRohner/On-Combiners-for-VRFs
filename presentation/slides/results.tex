\section{Robust Black-Box DAG Combiners}

\begin{frame}
    \frametitle{Robust Black-Box DAG Combiners}

    \begin{block}{Definition (Robust $(n;k)$ Black-Box DAG Combiner --- informal)}
        Each algorithm of $\DAGComb{P}$ is given by a directed acyclic graph $\G$:
        \begin{itemize}
            \item source $s$, sink $t$, intermediate vertices labeled $\ell : V(\G) \to [n]$
            \item at each vertex $v$, a PPT machine $f^v$ computes a query $\mathbf q_v$ from predecessors' outputs and queries $\O_{\ell(v)}$
        \end{itemize}
        Execute in topological order; the sink emits the final output.
    \end{block}

    \only<2->{\emph{Why DAGs?} Decouples local computation from oracle queries --- structure becomes explicit.}
\end{frame}

    \begin{center}
        \begin{adjustbox}{max width=\textwidth, max totalheight=0.85\textheight}

                        \begin{subprocedure}
                        \dbox{\procedure{$f^{v_1}(\x_s)$}{
                            \pcreturn \x_s
                        }}
                        \end{subprocedure}\\

                        \pcreturn \x_{v_1}
                    }
                \end{pcvstack}
            };
            \node (v2) at (3,-4) {

                        }}
                        \end{subprocedure}\\

                        \pcreturn \x_{v_2}

                        }}
                        \end{subprocedure}\\

                    }
                \end{pcvstack}
            };

\begin{frame}
    \frametitle{Equivalence}

    \begin{itemize}
        \item<1-> Robust $(n;k)$ black-box DAG combiner $\implies$ robust $(n;k)$ black-box combiner
        \item<2-> Robust $(n;k)$ black-box combiner $\implies$ robust $(n;k)$ black-box DAG combiner
    \end{itemize}

    \vspace{1.5em}
    \begin{center}
        \only<3->{\textbf{It suffices to (im)prove the DAG version.}}
    \end{center}
\end{frame}

\section{Impossibility}

\begin{frame}[c]
    \frametitle{Main Theorem}

    \begin{center}
        \begin{minipage}{0.9\textwidth}
            \Large
            \raggedright
            \hyphenpenalty=10000\exhyphenpenalty=10000
            There exists no robust $(2;1)$ black-box VRF-combiner.
        \end{minipage}
    \end{center}
\end{frame}

\begin{frame}
    \frametitle{The Two Candidates}

    Suppose $\DAGComb{VRF}$ is a robust $(2;1)$ black-box DAG VRF-combiner.

    \vspace{1em}
    \begin{center}
    \begin{tikzpicture}[
        cand/.style={draw, rounded corners, minimum width=4.5cm, minimum height=2cm, align=center, inner sep=8pt}
    ]
        \node[cand, fill=hlblue]  (vrf)  at (0,0)   {\large $\mathcal O_{vrf}$\\[0.3em] secure VRF};
        \node[cand, fill=hlpeach] (uppc) at (6.5,0) {\large $\mathcal O_{uppc}$\\[0.3em] trivial VF};
    \end{tikzpicture}
    \end{center}

    \vspace{0.4em}
    \only<2->{$\mathcal O_{uppc}$: every $\alg{Eval}{}(\sk, x)$ returns the same default $y_{def}$.}

    \vspace{0.4em}
    \only<3->{$\mathcal O_{vrf}$ secure $\Rightarrow$ $\DAGComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]$ is a secure VRF.}

    \vspace{0.4em}
    \only<4->{\textbf{Goal:} adversary $\mathcal A^{\textsf{PSPACE}}$ breaking its pseudorandomness.}
\end{frame}

\begin{frame}
    \frametitle{The Idea}

    The pseudorandomness experiment hides a secret key $\sk^* \in \SK = \bin^{s(\lambda)}$.

    \vspace{0.8em}
    \begin{center}
    \begin{tikzpicture}
        \draw[thick, rounded corners] (0,0) rectangle (8.5, 2.4);
        \node[align=center] at (1.3, 1.2) {$\SK$\\[0.2em]\small $2^{s(\lambda)}$};
        \fill[acgred] (5.4, 1.1) circle (3pt);
        \node[anchor=west] at (5.55, 1.1) {\small $\sk^*$};

        \only<2->{
            \draw[thick, rounded corners, fill=hlblue, fill opacity=0.55] (4.9, 0.5) rectangle (6.7, 1.7);
            \node at (5.8, 0.8) {\small $\SK_{cand}$};
        }
    \end{tikzpicture}
    \end{center}

    \vspace{0.3em}
    \only<2->{Use the $\textsf{PSPACE}$ oracle to \emph{filter} $\SK$ down to a small $\SK_{cand}$ that still contains $\sk^*$.}

    \vspace{0.4em}
    \only<3->{If $\abs{\SK_{cand}} \in \poly$, sampling uniformly gives $\sk_{cand} = \sk^*$ with non-negligible probability.}
\end{frame}

\begin{frame}
    \frametitle{$\alg{EvalSim}{VRF}$ as a Black Box}

    Assume an algorithm $\alg{EvalSim}{VRF}(\sk, x, y, \pi, \tau) \in \{\top, \bot\}$ with the following properties:

    \vspace{0.6em}
    \begin{itemize}\setlength\itemsep{0.5em}
        \item<2-> \textbf{makes no oracle queries}
        \item<3-> returns $\top$ on the experiment's secret key $\sk^*$
        \item<4-> returns $\bot$ when $\sk$'s combiner output disagrees with $y$
        \item<5-> runs in polynomial space
    \end{itemize}

    \vspace{0.8em}
    \only<6->{\emph{Constructing $\alg{EvalSim}{VRF}$ is the technical core of the thesis.}}

    \vspace{0.4em}
    \only<7->{The \emph{no-oracle-queries} property is what makes everything else possible.}
\end{frame}

\begin{frame}
    \frametitle{Constructing $\alg{EvalSim}{VRF}$}

    Traverse the DAG in topological order. At each oracle-call vertex:

    \vspace{0.5em}
    \begin{center}
    \begin{tikzpicture}[
        every node/.style={font=\small},
        box/.style={draw, rounded corners, align=center, inner sep=6pt}
    ]
        \node[box]              (root)     at (0, 3.6)   {oracle call at vertex $v$};
        \node[box, fill=hlpeach](uppc)     at (-4.6, 1.8){reply $y_{def}$};
        \node[box]              (vrf)     at (2.5, 1.8) {check trace $\tau$};
        \node[box, fill=hlblue] (intau)   at (0, 0)     {reply $\tau(\mathbf q)$};
        \node[box, fill=hlgreen](notintau) at (5, 0)    {branch over all\\well-formed outputs};
        \draw[->, thick] (root) -- (uppc)     node[midway, above, sloped, font=\footnotesize]{$\mathcal O_{uppc}$};
        \draw[->, thick] (root) -- (vrf)      node[midway, above, sloped, font=\footnotesize]{$\mathcal O_{vrf}$};
        \draw[->, thick] (vrf)  -- (intau)    node[midway, above, sloped, font=\footnotesize]{$\mathbf q \in \tau$};
        \draw[->, thick] (vrf)  -- (notintau) node[midway, above, sloped, font=\footnotesize]{$\mathbf q \notin \tau$};
    \end{tikzpicture}
    \end{center}

    \vspace{0.4em}
    Return $\top$ iff \emph{every} branch produces the honest image $y$.
\end{frame}

    Enumerate all $\sk \in \SK$ via DFS over a binary tree of depth $s(\lambda)$; run $\alg{EvalSim}{VRF}$ at each leaf.

    \begin{itemize}\setlength\itemsep{0.3em}
        \item<2-> DFS stack: depth $s(\lambda)$
        \item<3-> $\alg{EvalSim}{VRF}$ tape: polynomial, \emph{reused} at each leaf
    \end{itemize}

    \vspace{0.3em}
    \only<4->{Total memory polynomial $\Rightarrow$ in $\textsf{PSPACE}$.}

\begin{frame}
    \frametitle{Two Cases After Filtering}

    \vspace{0.4em}
    \begin{center}
    \begin{tikzpicture}[
        node distance=1.2cm,
        case/.style={draw, rounded corners, minimum width=4.6cm, minimum height=2cm, align=center, inner sep=8pt}
    ]
        \node (top) at (0, 0.8) {\textit{after filtering with all $x_i$}};
        \node[case, fill=hlgreen]  (a) at (-3.3, -1.6) {\textbf{Case A}\\[0.3em] $\abs{\SK_{cand}} \in \poly$};
        \node[case, fill=hlpeach]  (b) at ( 3.3, -1.6) {\textbf{Case B}\\[0.3em] $\SK_{cand}$ stays large};
        \draw[thick, ->] (top) -- (a);
        \draw[thick, ->] (top) -- (b);
    \end{tikzpicture}
    \end{center}

    \vspace{0.3em}
    \only<2->{\textbf{A:} sample $\sk_{cand}$, \emph{guess} $\sk_{cand} = \sk^*$.}

    \vspace{0.3em}
    \only<3->{\textbf{B:} many $\sk_{cand}$ collide with $\sk^*$ on random points; \emph{evaluate} $x^*$ under $\sk_{cand}$.}
\end{frame}

\begin{frame}[c]
    \frametitle{The Adversary}

    {\large
    \begin{enumerate}
        \setlength\itemsep{0.8em}
        \item<1-> Sample $x_1, \ldots, x_{s+\lambda} \sample \bin^\lambda$.
        \item<2-> Query the eval oracle: $(y_i, \pi_i) \gets \mathcal O_\text{eval}(x_i)$.
        \item<3-> Compute the verification trace
        \only<3->{%
        \vspace{0.9em}
        \[
            \tau_i \gets \text{trace}\bigl(\DAGComb{VRF}.\alg{Vfy}{VRF}(\pk^*, x_i, y_i, \pi_i)\bigr).
        \]
        \vspace{0.4em}
        }
        \item<4-> Define the candidate-key set
        \only<4->{%
        \vspace{0.9em}
        \[
            \SK_{cand} \coloneqq \{\sk : \alg{EvalSim}{VRF}(\sk, x_i, y_i, \pi_i, \tau_i) = \top \text{ for all } i\}
        \]
        \vspace{0.4em}

        and use $\textsf{PSPACE}$ to sample $\sk_{cand} \sample \SK_{cand}$.
        }
        \item<5-> Apply the strategy of Case A or Case B.
    \end{enumerate}
    }
\end{frame}

\begin{frame}[c]
    \frametitle{The Bound}

    \begin{center}
        $\Adv{\DAGComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]}{prf}{\lambda, \mathcal A} \;\geq\; \dfrac{1}{2} - \dfrac{1}{2^{\lambda+1}} - \dfrac{1}{\abs{\Y}}$
    \end{center}

    \vspace{1.2em}
    \begin{center}
        Non-negligible \quad$\Rightarrow$\quad contradicts pseudorandomness. \quad$\Box$
    \end{center}
\end{frame}

\section{Extensions}

\begin{frame}
    \frametitle{From $(2;1)$ to $(n;\lfloor n/2 \rfloor)$: Rotating Instances}

    Given a robust $(n;\lfloor n/2 \rfloor)$ black-box VRF-combiner $\BBComb{VRF}$, build a $(2;1)$ combiner $\DAGComb{VRF}$:
    \begin{itemize}
        \item<2-> Initialize the first $\lfloor n/2 \rfloor$ candidate slots with $\O_1$
        \item<3-> Initialize the remaining $\lceil n/2 \rceil$ slots with $\O_2$
    \end{itemize}

    \vspace{0.6em}
    \only<4->{If \emph{either} $\O_1$ or $\O_2$ is secure, at least $\lfloor n/2 \rfloor$ slots are secure $\Rightarrow$ $\BBComb{VRF}$ is secure.}

    \vspace{0.4em}
    \only<5->{So $\DAGComb{VRF}$ is a robust $(2;1)$ combiner --- contradicting the impossibility above.}

    \vspace{0.4em}
    \only<6->{\emph{Contraposition:} no robust $(n;\lfloor n/2 \rfloor)$ black-box VRF-combiner.}
\end{frame}

\begin{frame}
    \frametitle{Results}

    \begin{block}{Theorem ($1$-out-of-$2$ impossibility)}
        There is no robust $(2;1)$ black-box VRF-combiner.
    \end{block}

    \begin{block}{Corollary ($\lfloor n/2 \rfloor$-out-of-$n$ impossibility)}
        There is no robust $(n;\lfloor n/2 \rfloor)$ black-box VRF-combiner.
    \end{block}

    \begin{block}{Conjecture ($(n-1)$-out-of-$n$ impossibility)}
        There is no robust $(n;n-1)$ black-box VRF-combiner.
    \end{block}

    \vspace{0.4em}
    \only<2->{The same proof yields impossibility for VUF and wVRF.}
\end{frame}

\section{Beyond Impossibility}

\begin{frame}
    \frametitle{Stronger Candidate Assumptions}

    What if all candidates are \emph{uniquely provable} \emph{and} \emph{perfectly correct}?

    \begin{itemize}
        \item<2-> Both are \emph{information-theoretic} properties --- typically proven, not assumed.
        \item<3-> No need to hedge against them with combiners.
    \end{itemize}

    \vspace{0.8em}
    \only<4->{Under these assumptions, a simple robust $(n;1)$ black-box VRF-combiner exists.}
\end{frame}

\begin{frame}
    \frametitle{Robust $(n;1)$ Black-Box UP-VRF-Combiner}

    \begin{center}
        \begin{adjustbox}{max width=\textwidth}
            \begin{pchstack}[space=1em]
                \procedure[linenumbering]
                {$\BBComb{VRF}[\O_1, \O_2].\alg{Gen}{VRF}(1^\lambda)$}
                {
                    (\sk_1, \pk_1) \sample \O_1.\alg{Gen}{VRF}(1^\lambda)\\
                    (\sk_2, \pk_2) \sample \O_2.\alg{Gen}{VRF}(1^\lambda)\\
                    \pcreturn ((\sk_1, \sk_2), (\pk_1, \pk_2))
                }
                \procedure[linenumbering]
                {$\BBComb{VRF}[\O_1, \O_2].\alg{Eval}{VRF}(\sk, x)$}
                {
                    (\sk_1, \sk_2) \gets \sk\\
                    (y_1, \pi_1) \gets \O_1.\alg{Eval}{VRF}(\sk_1, x)\\
                    (y_2, \pi_2) \gets \O_2.\alg{Eval}{VRF}(\sk_2, x)\\
                    y \gets y_1 \xor y_2\\
                    \pi \gets (y_1, \pi_1, y_2, \pi_2)\\
                    \pcreturn (y, \pi)
                }
                \procedure[linenumbering]
                {$\BBComb{VRF}[\O_1, \O_2].\alg{Vfy}{VRF}(\pk, x, y, \pi)$}
                {
                    (\pk_1, \pk_2) \gets \pk\\
                    (y_1, \pi_1, y_2, \pi_2) \gets \pi\\
                    b_1 \gets \O_1.\alg{Vfy}{VRF}(\pk_1, x, y_1, \pi_1)\\
                    b_2 \gets \O_2.\alg{Vfy}{VRF}(\pk_2, x, y_2, \pi_2)\\
                    \pcreturn b_1 \land b_2 \land (y = y_1 \xor y_2)
                }
            \end{pchstack}
        \end{adjustbox}
    \end{center}

    \vspace{0.6em}
    \only<2->{Iterate pairwise to obtain a robust $(n;1)$ combiner.}
\end{frame}
