\begin{frame}
    \frametitle{Robust Black-Box DAG Combiners}

    \begin{block}{Definition (Robust $(n;k)$ Black-Box Combiner (Informal))}
        Robust $(n;k)$ black-box DAG $\prim{P}$-combiner $\DAGComb{P}$ consist for every algorithm:
        \begin{itemize}
            \item Directed acyclic graph $\G$
            \item Source $s \in V(\G)$
            \item Sink $t \in V(\G)$
            \item Vertex labeling $\ell : V(\G) \to \prim{P} \times [n]$
            \item PPT Turing machines $\{f^v\}_{v \in V(\G) \setminus \{s\}}$ 
        \end{itemize}
        Execute $\DAGComb{P}$ in topological order. At $v \in V(\G)$ evaluate $f^v$ on all predecessor outputs and query $\mathcal O_{\ell(v)}$.
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Equivalence}

    \begin{itemize}
        \item<1-> Robust $(n;k)$ black-box DAG Combiner $\implies$ Robust $(n;k)$ black-box combiners
        \item<2-> Robust $(n;k)$ black-box combiners $\implies$ Robust $(n;k)$ black-box DAG combiner
    \end{itemize}

    \vspace{1em}

    \begin{center}
        \only<3->{\textbf{Sufficient to show (im)possibility of a robust $(n;k)$ black-box DAG combiner.}}
    \end{center}
\end{frame}

\begin{frame}
    \frametitle{The $1$-out-of-$2$ Combiner}

    \begin{itemize}
        \item<1-> Simplest non-trivial case, only two candidates
        \item<2-> Construction extends to robust $1$-out-of-$n$ black-box combiner
        \item<3-> Impossibility extends to robust $\lfloor n/2\rfloor$-out-of-$n$ black-box combiner
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Setup}

    Suppose $\BBComb{VRF}$ is a robust $(2;1)$ black-box DAG VRF-combiner. The proof works by

    \begin{itemize}
        \item Candidates $\mathcal O_1, \mathcal O_2$ of which at least one is secure
        \item Adversary $\mathcal A$ against 
    \end{itemize}

    \begin{itemize}
        \item $\mathcal O_{vrf}$: a secure VRF
        \item $\mathcal O_{uppc}$: a \emph{trivial}, uniquely provable and perfectly correct VF:
            \begin{itemize}
                \item $\alg{Gen}{VRF}(1^\lambda) = (\sk_{def}, \pk_{def})$
                \item $\alg{Eval}{VRF}(\sk, x) = (y_{def}, \pi_{def})$
                \item $\alg{Vfy}{VRF}(\pk, x, y, \pi) = 1 \iff y = y_{def}$
            \end{itemize}
        \item $\mathcal O_{vrf}$ a secure VRF $\implies$ $\BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]$ a secure VRF
    \end{itemize}

    \vspace{0.3em}

    \textbf{Goal:} Adversary $\mathcal A^{\mathsf{PSPACE}, \mathcal O_{vrf}, \mathcal O_{uppc}}$
    against psuedorandomness of $\BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]$.

    \textbf{Assume}: $\SK = \bin^{s(\lambda)}$ for $s \in \poly(\lambda)$ for all candidates.
\end{frame}

\begin{frame}
    \frametitle{The Adversary}

    Assume we have a Turing machine (inefficient) $\alg{EvalSim}{VRF}$ satisfying:

    \begin{itemize}
        \item $\alg{EvalSim}{VRF}(\sk_{cand}, x, y, \pi, \tau) = \top \text{ if } \sk_{cand} = \sk$
        \item $\alg{EvalSim}{VRF}(\sk_{cand}, x, y, \pi, \tau) = \bot \text{ if } \alg{Eval}{VRF}^y(\sk_{cand}, x) \neq \alg{Eval}{VRF}^y(\sk, x)$
        \item $\alg{EvalSim}{VRF}$ makes \textit{no queries to any oracles}.
        \item $\alg{EvalSim}{VRF}$ has a \textit{polynomial space complexity}.
    \end{itemize}

    For $x \in \bin^\lambda, (y,\pi) = \BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]\alg{Eval}{VRF}(\sk, x)$ and $\tau = \text{trace}(\BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}].\alg{Vfy}{VRF}(\pk, x, y, \pi))$.
\end{frame}

\begin{frame}
    \frametitle{The Adversary}

    Adversary $\mathcal A_1^{\textsf{PSPACE}, \mathcal O_{vrf}, \mathcal O_{uppc}, \mathcal O_{\text{eval}}}(\pk)$:

    \begin{enumerate}
        \setlength{\itemsep}{1em}
        \item Sample $x_i \sample \bin^\lambda$ for all $i \in [s(\lambda) + \lambda]$
        \item For all $i \in [s(\lambda) + \lambda]$ query
        \[
            (y_i, \pi_i) \gets \mathcal O_{\text{eval}}(x_i) 
        \]
        \item For all $i \in [s(\lambda) + \lambda]$ compute the verification trace
        \[
            \tau_i \gets \text{trace}\left(\BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}].\alg{Vfy}{VRF}(\pk, x_i, y_i, \pi_i)\right)
        \]  
        \item Sample a candidate secret key $\sk_{cand}$ using the PSPACE-complete oracle
        \[
            \sk_{cand} \sample \left\{\sk \in \SK : \forall i \in [s(\lambda) + \lambda], \alg{EvalSim}{VRF}(\sk, x_i, y_i, \pi_i, \tau_i) = \top\right\}
        \]
    \end{enumerate}
\end{frame}


\begin{frame}
    \frametitle{The Adversary}


    \[
        \mathbf{Adv}^{\textup{prf}}_{\BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]}(\lambda, \mathcal A)
        \ge \tfrac{1}{2} - \tfrac{1}{2^{\lambda + 1}} - \tfrac{1}{|\Y|}.
    \]
\end{frame}

\begin{frame}
    \frametitle{How to get $\alg{EvalSim}{VRF}$}

    Given $(\sk_{cand}, x, y, \pi, \tau)$, $\alg{EvalSim}{VRF}$ is computed by:

    \begin{itemize}
        \item Follow the DAG $\G_{\text{Eval}}$, describing $\BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]$, in some topological ordering
        \item For every vertex $v$ corresponding to a query to $\mathcal O_{uppc}$ answer with default value
        \item For every vertex $v$ corresponding to a query to $\mathcal O_{vrf}$ check $\tau$:
        \begin{itemize}
            \item If $\mathbf q \in \tau$: answer accordingly
            \item If $\mathbf q \notin \tau$: branch for every possible answer
        \end{itemize}
        \item Return $\top$ iff all branches produce the image $y$
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{How to get $\alg{EvalSim}{VRF}$}

    Given $(\sk_{cand}, x, y, \pi, \tau)$, $\alg{EvalSim}{VRF}$ is computed by:

    \begin{itemize}
        \item Follow the DAG $\G_{\text{Eval}}$, describing $\BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]$, in some topological ordering
        \item For every vertex $v$ corresponding to a query to $\mathcal O_{uppc}$ answer with default value
        \item For every vertex $v$ corresponding to a query to $\mathcal O_{vrf}$ check $\tau$:
        \begin{itemize}
            \item If $\mathbf q \in \tau$: answer accordingly
            \item If $\mathbf q \notin \tau$: branch for every possible answer
        \end{itemize}
        \item Return $\top$ iff all branches produce the image $y$
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{How to get $\alg{EvalSim}{VRF}$}

    Assume we have a Turing machine (inefficient) $\alg{EvalSim}{VRF}$ satisfying:

    \begin{itemize}
        \item $\alg{EvalSim}{VRF}(\sk_{cand}, x, y, \pi, \tau) = \top \text{ if } \sk_{cand} = \sk$
        \item $\alg{EvalSim}{VRF}(\sk_{cand}, x, y, \pi, \tau) = \bot \text{ if } \alg{Eval}{VRF}^y(\sk_{cand}, x) \neq \alg{Eval}{VRF}^y(\sk, x)$
        \item $\alg{EvalSim}{VRF}$ makes \textit{no queries to any oracles}.
        \item $\alg{EvalSim}{VRF}$ has a \textit{polynomial space complexity}.
    \end{itemize}

    For $x \in \bin^\lambda, (y,\pi) = \BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]\alg{Eval}{VRF}(\sk, x)$ and $\tau = \text{trace}(\BBComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}].\alg{Vfy}{VRF}(\pk, x, y, \pi))$.
\end{frame}

\begin{frame}
    \frametitle{How to get $\alg{EvalSim}{VRF}$}

    $\alg{EvalSim}{VRF}$ has a \textit{polynomial space complexity}:

    \begin{itemize}
        \item $\alg{EvalSim}{VRF}$ in $\textsf{NPSPACE}$ since every branch is an execution of $\alg{Eval}{VRF}$
        \item $\textsf{NPSPACE} \subseteq \textsf{PSPACE}$
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{How to get $\alg{EvalSim}{VRF}$}

    $\alg{EvalSim}{VRF}(\sk_{cand}, x, y, \pi, \tau) = \bot \text{ if } \alg{Eval}{VRF}^y(\sk_{cand}, x) \neq y$:

    \begin{itemize}
        \item All queries made to $\mathcal O_{uppc}$ answered the same $\alg{Eval}{VRF}(\sk_{cand}, x)$
        \item All queries made to $\mathcal O_{vrf}$ with $\mathbf q \in \tau$ answered the same
        \item For all queries made to $\mathcal O_{vrf}$ with $\mathbf q \notin \tau$ there exists a branch which is the same
    \end{itemize}
    
    $\implies$ At least one branch evaluates to $\alg{Eval}{VRF}^y(\sk_{cand}, x) \neq y$.
\end{frame}

\begin{frame}
    \frametitle{How to get $\alg{EvalSim}{VRF}$}

    $\alg{EvalSim}{VRF}(\sk_{cand}, x, y, \pi, \tau) = \top \text{ if } \sk_{cand} = \sk$.

    \vspace{.5em}

    \begin{itemize}
        \item Suppose $\alg{EvalSim}{VRF}(\sk, x, y, \pi, \tau) = \bot$
        \item Then there exist answers to queries to $\mathcal O_{vrf}$ such that the output is $y' \neq y$
        \item Modify $\mathcal O_{uppc}$ to make it a secure VRF such that $\mathcal O_{uppc}^*\big|_{\widetilde\SK} = \mathcal O_{uppc}\big|_{\widetilde\SK}$
        \item Modify $\mathcal O_{vrf}$ to produce the transcript of the branch, $\mathcal O_{vrf}^*$
        \item $\mathcal O_{vrf}$ and $\mathcal O_{vrf}^*$ only differ on queries $\mathbf q \notin \tau$ and indistinguishable for verification
    \end{itemize}

    Now we have:

    \begin{align*}
        y = \BBComb{VRF}[\mathcal O_{uppc}^*, \mathcal O_{vrf}].\alg{Eval}{VRF}^y(\sk, x) \neq 
        \BBComb{VRF}[\mathcal O_{uppc}^*, \mathcal O_{vrf}^*].\alg{Eval}{VRF}^y(\sk, x) = y^*
    \end{align*}
\end{frame}

\begin{frame}
    \frametitle{How to get $\alg{EvalSim}{VRF}$}

    $\alg{EvalSim}{VRF}(\sk_{cand}, x, y, \pi, \tau) = \top \text{ if } \sk_{cand} = \sk$.

    Now we have:

    \begin{align*}
        y = \BBComb{VRF}[\mathcal O_{uppc}^*, \mathcal O_{vrf}].\alg{Eval}{VRF}^y(\sk, x) \neq 
        \BBComb{VRF}[\mathcal O_{uppc}^*, \mathcal O_{vrf}^*].\alg{Eval}{VRF}^y(\sk, x) = y^*
    \end{align*}

    Since the verification algorithms are however the same on queries $\mathbf q \in \tau$, the proof $\pi$ verifies identically in both executions.

    \vspace{.5em}

    $\implies$ Two distinct values $y \neq y^*$ both verify under the same $(\pk, x, \pi)$, breaking \textbf{unique provability} of $\BBComb{VRF}$.
\end{frame}

\begin{frame}
    \frametitle{From $(2;1)$ to $(n;\lfloor n/2\rfloor)$}

    Given a robust $(n;\lfloor n/2\rfloor)$ black-box VRF-combiner $\BBComb{VRF}$, build a robust $(2;1)$ combiner $\DAGComb{VRF}$:

    \begin{itemize}
        \item initialize the first $\lfloor n/2\rfloor$ candidates of $\BBComb{VRF}$ with $\mathcal O_1$
        \item initialize the remaining $\lceil n/2\rceil$ candidates with $\mathcal O_2$
    \end{itemize}

    If either $\mathcal O_1$ or $\mathcal O_2$ is a secure VRF,
    at least $\lfloor n/2\rfloor$ candidates of $\BBComb{VRF}$ are secure, so $\DAGComb{VRF}$ is a robust $(2;1)$ combiner.

    \vspace{0.3em}
    Contraposition with the main theorem yields the impossibility of robust $(n;\lfloor n/2\rfloor)$ black-box VRF-combiners.
\end{frame}

\begin{frame}
    \frametitle{Result}
    \begin{block}{Theorem. Impossibility $1$-out-of-$2$ combiner.}
        There is \textbf{no} robust $(2;1)$ black-box VRF-combiner.\\
    \end{block}
    \begin{block}{Corollary. Impossibility $\lfloor n/2\rfloor$-out-of-$n$ combiner.}
        There is \textbf{no} robust $(n;\lfloor n/2\rfloor)$ black-box VRF-combiner.\\
    \end{block}
    \begin{block}{Conjecture. Impossibility $(n{-}1)$-out-of-$n$ combiner.}
        There is \textbf{no} robust $(n;n{-}1)$ black-box VRF-combiner.\\
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{What We Showed}

    Additionally the same proof shows that:

    \begin{block}{Lemma. Impossibility $\lfloor n/2\rfloor$-out-of-$n$ VUF-combiner.}
        There is \textbf{no} robust $(2;1)$ black-box VRF-combiner.\\
    \end{block}
    \begin{block}{Lemma. Impossibility $\lfloor n/2\rfloor$-out-of-$n$ wVRF-combiner.}
        There is \textbf{no} robust $(2;1)$ black-box VRF-combiner.\\
    \end{block}
\end{frame}
