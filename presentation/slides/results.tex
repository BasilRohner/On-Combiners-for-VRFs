\section{Robust Black-Box DAG Combiners}

\begin{frame}
    \frametitle{Robust Black-Box DAG Combiners}

    \begin{block}{Definition (Robust $(n;k)$ Black-Box DAG Combiner, informal)}
        A DAG $\G$ where each vertex $v$ runs a PPT $f^v$ and queries one candidate $\O_{\ell(v)}$; topological traversal produces the algorithm's output.

        \smallskip
        \emph{Robust $(n;k)$:} the induced oracle TM is a robust $(n;k)$ black-box $\prim{P}$-combiner.
    \end{block}

    \only<2->{\emph{Why DAGs?} Decouples \hlb{local computation} from \hlb{oracle queries}.}
\end{frame}

\section{Impossibility}

\begin{frame}[c]
    \frametitle{Main Theorem}

    \begin{center}
        \begin{minipage}{0.9\textwidth}
            \Large
            \raggedright
            \hyphenpenalty=10000\exhyphenpenalty=10000
            There exists no robust $(2;1)$ black-box VRF-combiner.
        \end{minipage}
    \end{center}
\end{frame}

\begin{frame}
    \frametitle{The Two Candidates}

    Suppose $\DAGComb{VRF}$ is a robust $(2;1)$ black-box DAG VRF-combiner.

    \vspace{1em}
    \begin{center}
    \begin{tikzpicture}[
        cand/.style={draw, rounded corners, minimum width=4.5cm, minimum height=2cm, align=center, inner sep=8pt}
    ]
        \node[cand, fill=hlblue]  (vrf)  at (0,0)   {\large $\mathcal O_{vrf}$\\[0.3em] secure VRF};
        \node[cand, fill=hlpeach] (uppc) at (6.5,0) {\large $\mathcal O_{uppc}$\\[0.3em] trivial VF};
    \end{tikzpicture}
    \end{center}

    \vspace{0.4em}
    \only<2->{$\mathcal O_{uppc}$: every $\alg{Eval}{}(\sk, x)$ returns the same default $y_{def}$.}

    \vspace{0.4em}
    \only<3->{$\mathcal O_{vrf}$ secure $\Rightarrow$ $\DAGComb{VRF}[\mathcal O_{vrf}, \mathcal O_{uppc}]$ is a secure VRF.}

    \vspace{0.4em}
    \only<4->{\textbf{Goal:} adversary $\mathcal A^{\textsf{PSPACE}}$ breaking its pseudorandomness.}
\end{frame}

\begin{frame}
    \frametitle{The Idea}

    The pseudorandomness experiment hides a secret key $\sk^* \in \SK = \bin^{s(\lambda)}$.

    \vspace{0.8em}
    \begin{center}
    \begin{tikzpicture}
        \draw[thick, rounded corners] (0,0) rectangle (8.5, 2.4);
        \node[align=center] at (1.3, 1.2) {$\SK$};
        \fill[acgred] (5.4, 1.1) circle (3pt);
        \node[anchor=west] at (5.55, 1.1) {\small $\sk^*$};

        \only<2->{
            \draw[thick, rounded corners, fill=hlblue, fill opacity=0.55] (4.9, 0.5) rectangle (6.7, 1.7);
            \node[anchor=south] at (5.8, 1.75) {\small $\SK_{cand}$};
        }
    \end{tikzpicture}
    \end{center}

    \vspace{0.3em}
    \only<2->{Use the $\textsf{PSPACE}$ oracle to \emph{filter} $\SK$ down to a small $\SK_{cand}$ that still contains $\sk^*$.}

    \vspace{0.4em}
    \only<3->{If $\abs{\SK_{cand}} \in \poly$, sampling uniformly gives $\sk_{cand} = \sk^*$ with non-negligible probability.}
\end{frame}

\begin{frame}
    \frametitle{$\alg{EvalSim}{VRF}$ as a Black Box}

    Assume an algorithm $\alg{EvalSim}{VRF}(\sk, x, y, \pi, \tau) \in \{\top, \bot\}$ with the following properties:

    \vspace{0.6em}
    \begin{itemize}\setlength\itemsep{0.5em}
        \item<2-> makes no oracle queries
        \item<3-> returns $\top$ on the experiment's secret key $\sk^*$
        \item<4-> returns $\bot$ when $\sk$'s combiner output disagrees with $y$
        \item<5-> runs in polynomial space
    \end{itemize}

    \vspace{0.8em}
    \only<6->{\emph{Constructing $\alg{EvalSim}{VRF}$ and proving these properties hold is the technical core of the thesis.}}

\end{frame}

\begin{frame}
    \frametitle{Constructing $\alg{EvalSim}{VRF}$}

    Traverse the DAG in topological order. At each oracle-call vertex:

    \vspace{0.5em}
    \begin{center}
    \begin{tikzpicture}[
        every node/.style={font=\small, align=center, inner sep=4pt},
        elabel/.style={fill=white, inner sep=2pt, font=\footnotesize}
    ]
        \node              (root)     at (0, 3.6)   {oracle call at vertex $v$};
        \node              (uppc)     at (-4.6, 1.8){\hlb{reply $y_{def}$}};
        \node              (vrf)      at (2.5, 1.8) {check trace $\tau$};
        \node              (intau)    at (0, 0)     {\hlb{reply $\tau(\mathbf q)$}};
        \node              (notintau) at (5, 0)     {\hlr{branch over all well-formed outputs}};
        \draw[->, thick] (root) -- (uppc)     node[midway, elabel]{$\mathcal O_{uppc}$};
        \draw[->, thick] (root) -- (vrf)      node[midway, elabel]{$\mathcal O_{vrf}$};
        \draw[->, thick] (vrf)  -- (intau)    node[midway, elabel]{$\mathbf q \in \tau$};
        \draw[->, thick] (vrf)  -- (notintau) node[midway, elabel]{$\mathbf q \notin \tau$};
    \end{tikzpicture}
    \end{center}

    \vspace{0.4em}
    Return $\top$ iff \emph{every} branch produces the honest image $y$.
\end{frame}

\begin{frame}
    \frametitle{$\alg{EvalSim}{VRF}$ Revisited}

    Properties of $\alg{EvalSim}{VRF}(\sk, x, y, \pi, \tau)$:

    \vspace{0.8em}
    \begin{itemize}\setlength\itemsep{0.7em}
        \item[\alt<2->{\textcolor{acgred}{$\checkmark$}}{$\bullet$}] makes no oracle queries
        \item[$\bullet$] returns $\top$ on the experiment's secret key $\sk^*$
        \item[\alt<3->{\textcolor{acgred}{$\checkmark$}}{$\bullet$}] returns $\bot$ when $\sk$'s combiner output disagrees with $y$
        \item[\alt<4->{\textcolor{acgred}{$\checkmark$}}{$\bullet$}] runs in polynomial space
    \end{itemize}
\end{frame}

\begin{frame}[c]
    \frametitle{The Adversary}

    \begin{enumerate}
        \setlength\itemsep{0.6em}
        \item<1-> Sample $x_1, \ldots, x_{s+\lambda} \sample \bin^\lambda$ \quad (recall $\SK = \bin^{s(\lambda)}$).
        \item<2-> Query the eval oracle: $(y_i, \pi_i) \gets \mathcal O_\text{eval}(x_i)$.
        \item<3-> Compute the verification traces $\tau_i \gets \text{trace}\bigl(\DAGComb{VRF}[\O_{vrf}, \O_{uppc}].\alg{Vfy}{}(\pk^*, x_i, y_i, \pi_i)\bigr)$.
        \item<4-> Use $\textsf{PSPACE}$ to sample $\sk_{cand} \sample \SK_{cand}$, where
        \only<4->{%
        \vspace{0.2em}
        \[
            \SK_{cand} \coloneqq \big\{\sk \in \SK : \alg{EvalSim}{VRF}(\sk, x_i, y_i, \pi_i, \tau_i) = \top \;\forall i\big\}.
        \]
        }
        \item<5-> Sample challenge $x^* \sample \bin^\lambda$ and obtain $y^*$.
        \item<6-> Check $\DAGComb{VRF}[\O_{vrf}, \O_{uppc}].\alg{Eval}{}(\sk_{cand}, x^*) \overset{?}{=} (y^*, \pi)$.
    \end{enumerate}
\end{frame}

\begin{frame}[c]
    \frametitle{Bounding the Advantage}

    \begin{align*}
        \Adv{\DAGComb{VRF}[\O_{vrf}, \O_{uppc}]}{prf}{\lambda, \mathcal A}
        &= 2 \cdot \bigl|\Pr{\mathcal A \text{ wins}} - \tfrac{1}{2}\bigr|
        \only<2->{\\ &\geq 2 \cdot \bigl|\Pr{\sk_{cand} \text{ and } \sk^* \text{ produce the same eval image on fresh } x^*} - \tfrac{1}{2}\bigr|}
        \only<3->{\\ &\geq \dfrac{1}{2} - \dfrac{1}{2^{\lambda+1}} - \dfrac{1}{\abs{\Y}} \quad \text{(by a Markov argument)}}
    \end{align*}

    \vspace{1em}
    \only<4->{
        \begin{center}
            $\Adv{\DAGComb{VRF}[\O_{vrf}, \O_{uppc}]}{prf}{\lambda, \mathcal A} \notin \negl(\lambda) \;\Rightarrow\;$ contradicts pseudorandomness. \quad $\Box$
        \end{center}
    }
\end{frame}

\section{Results}

\begin{frame}[c]
    \frametitle{Results}

    \begin{block}{Theorem}
        No robust $(2;1)$ black-box \alt<2->{VRF/VUF/wVRF}{VRF}-combiner.
    \end{block}

    \begin{block}{Corollary}
        No robust $(n;\lfloor n/2\rfloor)$ black-box \alt<2->{VRF/VUF/wVRF}{VRF}-combiner.
    \end{block}

    \begin{block}{Conjecture}
        No robust $(n;n-1)$ black-box \alt<2->{VRF/VUF/wVRF}{VRF}-combiner.
    \end{block}
\end{frame}

\begin{frame}[c]
    \frametitle{Extending to $(n;\lfloor n/2 \rfloor)$ Combiners}

    Fill the $n$ slots of an $(n;\lfloor n/2\rfloor)$ combiner with two oracles $\O_1, \O_2$.

    \vspace{0.8em}
    \only<2->{Security of either $\Rightarrow$ $\geq \lfloor n/2\rfloor$ slots secure $\Rightarrow$ a $(2;1)$ combiner.}

    \vspace{0.5em}
    \only<3->{Contraposition $\Rightarrow$ no $(n;\lfloor n/2\rfloor)$ black-box VRF-combiner.}
\end{frame}

\section{Beyond Impossibility}

\begin{frame}[c]
    \frametitle{Beyond Impossibility}

    If all candidates are \emph{uniquely provable} and \emph{perfectly correct} (information-theoretic, typically proven):

    \vspace{1em}
    \only<2->{a simple robust $(n;1)$ black-box VRF-combiner exists.}
\end{frame}

\begin{frame}
    \frametitle{Robust $(n;1)$ Black-Box UP-VRF-Combiner}

    \begin{center}
        \begin{adjustbox}{max width=\textwidth}
            \begin{pchstack}[space=1em]
                \procedure[linenumbering]
                {$\BBComb{VRF}[\O_1, \O_2].\alg{Eval}{VRF}(\sk, x)$}
                {
                    (\sk_1, \sk_2) \gets \sk\\
                    (y_1, \pi_1) \gets \O_1.\alg{Eval}{VRF}(\sk_1, x)\\
                    (y_2, \pi_2) \gets \O_2.\alg{Eval}{VRF}(\sk_2, x)\\
                    y \gets y_1 \xor y_2\\
                    \pi \gets (y_1, \pi_1, y_2, \pi_2)\\
                    \pcreturn (y, \pi)
                }
                \procedure[linenumbering]
                {$\BBComb{VRF}[\O_1, \O_2].\alg{Vfy}{VRF}(\pk, x, y, \pi)$}
                {
                    (\pk_1, \pk_2) \gets \pk\\
                    (y_1, \pi_1, y_2, \pi_2) \gets \pi\\
                    b_1 \gets \O_1.\alg{Vfy}{VRF}(\pk_1, x, y_1, \pi_1)\\
                    b_2 \gets \O_2.\alg{Vfy}{VRF}(\pk_2, x, y_2, \pi_2)\\
                    \pcreturn b_1 \land b_2 \land (y = y_1 \xor y_2)
                }
            \end{pchstack}
        \end{adjustbox}
    \end{center}

    \vspace{0.6em}
    \only<2->{Iterate pairwise to obtain a robust $(n;1)$ combiner.}
\end{frame}
