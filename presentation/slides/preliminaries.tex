\section{Preliminaries}

\begin{frame}
    \frametitle{Verifiable Functions}
    \begin{block}{Definition (Verifiable Function)}
        A \emph{verifiable function} (VF) $\prim{VF}$ is a triple of the following efficient algorithms:
        \begin{itemize}
            \item $\alg{Gen}{VF}(1^\lambda) = (\sk, \pk)$ generates a key-pair
            \item $\alg{Eval}{VF}(\sk, x) = (y, \pi)$ evaluates points to an image-proof pair
            \item $\alg{Vfy}{VF}(\pk, x, y, \pi) = b$ verifies evaluations
        \end{itemize}
    \end{block}
    \vspace{0.3em}
    \only<2->{A VF is a purely syntactic object, we define security properties on top of it.}
\end{frame}

\begin{frame}
    \frametitle{Properties of Verifiable Functions}
    \begin{block}{$\varepsilon$-Correctness}
        Honestly generated image-proof pairs verify with probability at least $\varepsilon$:
        \[
            \Pr{
                \begin{aligned}
                    (\sk, \pk) &\sample \alg{Gen}{VF}(1^\lambda)\\ 
                    x &\sample \bin^\lambda\\
                    (y, \pi) &\gets \alg{Eval}{VF}(\sk, x)
                \end{aligned} :
                \alg{Vfy}{VF}(\pk, x, y, \pi) = 1 
            } \ge \varepsilon(\lambda)
        \]
        $\prim{VF}$ is \emph{correct} if it is $1$-correct and \emph{statistically correct} if it is $(1-\varepsilon)$-correct and $\varepsilon \in \negl(\lambda)$.
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Properties of Verifiable Functions}
    \begin{block}{Unique Provability}
        For every public key $\pk$ and every point $x$, at most one image $y$ is provable:
        \[
            \alg{Vfy}{VRF}(\pk, x, y_0, \pi_0) = \alg{Vfy}{VRF}(\pk, x, y_1, \pi_1) = 1
            \;\Longrightarrow\; y_0 = y_1.
        \]
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Properties of Verifiable Functions}
    \begin{block}{$(\varepsilon, \frakt)$-Pseudorandomness}
        For any adversary $\mathcal A$ running in time at most $\frakt$:
        \[
            \Adv{VF}{prf}{\lambda, \mathcal A} := 2 \cdot \abs*{\Pr{\Exp{VF}{prf}(\lambda, \mathcal A) = 1} - \frac{1}{2}} \le \varepsilon(\lambda)
        \]
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Properties of Verifiable Functions}
    \begin{block}{$(\varepsilon, \frakt)$-Weak Pseudorandomness}
        For any adversary $\mathcal A$ running in time at most $\frakt$ in the wprf game:
        \[
            \Adv{VF}{wprf}{\lambda, \mathcal A} := 2 \cdot \abs*{\Pr{\Exp{VF}{wprf}(\lambda, \mathcal A) = 1} - \frac{1}{2}} \le \varepsilon(\lambda)
        \]
        \emph{Weakly pseudorandom} if $\varepsilon \in \negl(\lambda)$ for $\frakt \in \poly(\lambda)$.
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Properties of Verifiable Functions}
    \begin{block}{$(\varepsilon, \frakt)$-Unpredictability}
        For any adversary $\mathcal A$ running in time at most $\frakt$ in the up game:
        \[
            \Adv{VF}{up}{\lambda, \mathcal A} := \Pr{\Exp{VF}{up}(\lambda, \mathcal A) = 1} \le \varepsilon(\lambda)
        \]
        \emph{Unpredictable} if $\varepsilon \in \negl(\lambda)$ for $\frakt \in \poly(\lambda)$.
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Verifiable Random Functions}
    \begin{block}{Verifiable Random Function}
        A \emph{verifiable random function} (VRF) is a VF that is:
        \begin{itemize}
            \item Correct
            \item Uniquely Provable
            \item Pseudorandom
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Weak Verifiable Random Functions}
    \begin{block}{Weak Verifiable Random Function}
        A \emph{weak verifiable random function} (wVRF) is a VF that is:
        \begin{itemize}
            \item Correct
            \item Uniquely Provable
            \item Weakly Pseudorandom
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Verifiable Unpredictable Functions}
    \begin{block}{Verifiable Unpredictable Function}
        A \emph{verifiable unpredictable function} (VUF) is a VF that is:
        \begin{itemize}
            \item Correct
            \item Uniquely Provable
            \item Unpredictable
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Robust Black-Box Combiners}
    \begin{block}{Definition (Robust $(n;k)$ Black-Box Combiner (Informal))}
        Robust $(n;k)$ black-box $\prim{P}$-combiner $\BBComb{P}$ satisfy:
        \begin{itemize}
            \item Give oracle access to $n$ candidate schemes $\mathcal O_1, \ldots, \mathcal O_n$
            \item Implements primitive $\prim{P}$
            \item Is secure if at least $k$ candidates are secure \only<2->{\textbf{even against \alt<3->{adversaries with a $\textsf{PSPACE}$ oracle}{inefficient adversaries}}}
        \end{itemize}
    \end{block}
\end{frame}
