\section{Preliminaries}

\begin{frame}
    \frametitle{Verifiable Functions}
    \begin{block}{Definition (Verifiable Function)}
        A \emph{verifiable function} (VF) $\prim{VF}$ is a triple of the following efficient algorithms:
        \begin{itemize}
            \item $\alg{Gen}{VF}(1^\lambda) = (\sk, \pk)$ generates a key-pair
            \item $\alg{Eval}{VF}(\sk, x) = (y, \pi)$ evaluates points to an image-proof pair
            \item $\alg{Vfy}{VF}(\pk, x, y, \pi) = b$ verifies evaluations
        \end{itemize}
    \end{block}
    \vspace{0.3em}
    A VF is a purely syntactic object, we define security properties on top of it.
\end{frame}

\begin{frame}
    \frametitle{Properties of Verifiable Functions}
    \begin{block}{$\varepsilon$-Correctness}
        Honestly generated image-proof pairs verify with probability at least $\varepsilon$:
        \[
            \Pr{
                \begin{aligned}
                    (\sk, \pk) &\sample \alg{Gen}{VF}(1^\lambda)\\ 
                    x &\sample \bin^\lambda\\
                    (y, \pi) &\gets \alg{Eval}{VF}(\sk, x)
                \end{aligned} :
                \alg{Vfy}{VF}(\pk, x, y, \pi) = 1 
            } \ge \varepsilon(\lambda)
        \]
        $\prim{VF}$ is \emph{correct} if it is $1$-correct and \emph{statistically correct} if it is $(1-\varepsilon)$-correct and $\varepsilon \in \negl(\lambda)$.
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Properties of Verifiable Functions}
    \begin{block}{Unique Provability}
        For every $\pk$ and every $x$, at most one image $y$ is provable:
        \[
            \alg{Vfy}{VRF}(\pk, x, y_0, \pi_0) = \alg{Vfy}{VRF}(\pk, x, y_1, \pi_1) = 1
            \;\Longrightarrow\; y_0 = y_1.
        \]
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Properties of Verifiable Functions}
    \begin{block}{$(\varepsilon, \frakt)$-Pseudorandom}
        For any adversary $\mathcal A$ running in time at most $\frakt$, playing the prf experiment $\Exp{VF}{prf}(\lambda, \mathcal A)$:
        


        The advantage $\Adv{VF}{prf}{\lambda, \mathcal A}$ is bounded from above by 
        \[
            \Adv{VF}{prf}{\lambda, \mathcal A} := 2 \cdot \abs*{\Pr{\Exp{VF}{prf}(\lambda, \mathcal A) = 1} - \frac{1}{2}} \le \varepsilon(\lambda)
        \]
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Verifiable Random Functions}
    \begin{block}{Verifiable Random Function}
        A \emph{verifiable random function} (VRF) is a VF that is:
        \begin{itemize}
            \item Correct
            \item Uniquely Provable
            \item Pseudorandom
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Robust Black-Box Combiners}
    \begin{block}{Definition (Robust $(n;k)$ Black-Box $\prim{P}$-Combiner)}
        A ppt.\ oracle Turing machine $\BBComb{P}$ that, given oracle access
        to $n$ candidates $\mathcal O_1, \ldots, \mathcal O_n$, implements primitive $\prim{P}$
        securely whenever at least $k$ of the candidates do s.t. computational security properties 
        hold against inefficient adversaries.
    \end{block}

    \begin{itemize}
        \item \textbf{Black-box:} the combiner queries candidates as oracles ---
            it cannot inspect or modify their code.
        \item We require security against adversaries with access to a
            \textbf{PSPACE oracle}, forcing the combiner to derive its security from the candidates.
        \item Known results: simple $(n;1)$ black-box combiners exist for PRFs,
            OWFs, PKE, key agreement, \ldots\ --- but \textbf{not known for VRFs}.
    \end{itemize}
\end{frame}
