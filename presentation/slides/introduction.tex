\section{Introduction}

\begin{frame}[c]
    \frametitle{Cryptographic Hardness Assumptions}

    {\large
        Most cryptographic constructions are built on a \emph{hardness assumption}.
        \par
    }

    \vspace{1.2em}

    \begin{itemize}\setlength{\itemsep}{0.6em}
        \item<2-> Factoring, Diffie--Hellman, LWE, isogenies, \ldots
        \item<3-> Each one \emph{might} fail: cryptanalysis, quantum computers, \ldots
        \item<4-> A broken assumption breaks every construction that relies on it
    \end{itemize}

    \vspace{2em}

    \only<5->{%
        \begin{center}
            \Large
            Don't bet on a \hlb{single} assumption.\\[0.25em]
            \emph{Combine} candidates based on \hlb{different} ones.
        \end{center}%
    }
\end{frame}

\begin{frame}[c]
    \frametitle{Research Question}

    \begin{center}
        \begin{minipage}{0.9\textwidth}

        \end{minipage}
    \end{center}
\end{frame}
