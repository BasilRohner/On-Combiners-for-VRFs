\section{Introduction}
\label{sec:intro}

\begin{frame}
    \frametitle{Research Question}
    Can we \emph{combine} multiple instances of verifiable random function (VRFs)
    into a single VRF that is secure as long as \emph{enough} of the instances are?

    \vspace{0.8em}

    \begin{itemize}
        \item How do we \emph{combine} the candidate schemes?
        \item What are \emph{enough} candidates to combine?
    \end{itemize}

    \vspace{0.8em}

    Can we combine $k$ out of $n$ VRF instances in a \emph{black-box} fashion?
\end{frame}
