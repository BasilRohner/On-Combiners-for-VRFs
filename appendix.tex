\subsection{Proof of \Cref{lemma:difference_lemma}}

We present the proof of the difference lemma, formally:

\begin{restatable}[Difference Lemma]{lemma}{difflemma}\label{lemma:difference_lemma}
    Let $\mathcal W_0, \mathcal W_1, \mathcal E$ be events with
    \begin{align}
        \Prob{\mathcal W_0 \land \lnot \mathcal E} = \Prob{\mathcal W_1 \land \lnot \mathcal E} \label{eq:dif}
    \end{align}
    Then we can bound the difference in probability of $\mathcal W_0$ and $\mathcal W_1$ with
    \[
        \abs*{\Prob{\mathcal W_0} - \Prob{\mathcal W_1}} \le \Prob{\mathcal E}
    \]
\end{restatable}

\paragraph{Proof of \Cref{lemma:difference_lemma}.}
Let $\mathcal W_0, \mathcal W_1, \mathcal E$ be events satisfying \eqref{eq:dif}. Then:
\begin{align}
    \abs*{\Prob{\mathcal W_0} - \Prob{\mathcal W_1}}
    =& 
    \big|\Prob{\mathcal W_0 \land \mathcal E} + \Prob{\mathcal W_0 \land \lnot \mathcal E} \\ 
    &\quad - \Prob{\mathcal W_1 \land \mathcal E} - \Prob{\mathcal W_1 \land \lnot \mathcal E}\big|\\ 
    =& \abs*{\Prob{\mathcal W_0 \land \mathcal E} - \Prob{\mathcal W_1 \land \mathcal E}}\\ 
    =& \abs*{\Prob{\mathcal W_0 \mid \mathcal E} \Prob{\mathcal E} - \Prob{\mathcal W_1 \mid \mathcal E} \Prob{\mathcal E}}\\ 
    =& \abs*{\Prob{\mathcal W_0 \mid \mathcal E} - \Prob{\mathcal W_1 \mid \mathcal E}} \Prob{\mathcal E}\\
    \le&\Prob{\mathcal E} 
\end{align}

\subsection{Proof of \Cref{claim:sk_iff_pk}}
\label{sec:sk_iff_pk}

\skiffpk*

\begin{proof}[\Cref{claim:sk_iff_pk}]
    We show $\sk \in \closure_\SK(\tildeSK, \tildePK) \implies \pk \in \closure_\PK(\tildeSK, \tildePK)$:

    By definition of the closure, we have:
    \[
    \sk \in \closure_\SK(\widetilde\SK) \lor \sk \in \bigcup_{\widetilde\pk \in \closure_\PK(\widetilde\PK)} R_\SK(\widetilde\pk)
    \]
    We distinguish two cases: First, assume $\sk \in \closure_\SK(\tildeSK)$, then:
    \[
        \pk \in R_\PK(\sk)
    \]
    Combining this with $\sk \in \closure_\SK(\tildeSK)$, we get:
    \[
        \pk \in \bigcup_{\widetilde\sk \in \closure_\SK(\tildeSK)} R_\PK(\widetilde\sk)
    \]
    And hence $\pk \in \closure_\PK(\tildeSK, \tildePK)$. 
    
    On the other hand, if $\sk \notin \closure_\SK(\tildeSK)$, then:
    \[
        \sk \in \bigcup_{\widetilde\pk \in \closure_\PK(\tildePK)} R_\SK(\widetilde\pk)
    \]
    In particular, we have $\sk \in R_\SK(\widetilde\pk)$ for some $\widetilde\pk$. By definition of $\sim_\PK$, we have
    $\widetilde\pk \sim_\PK \pk$ since $\sk$ is a secret key which can be sampled together with both $\pk$ and $\widetilde \pk$. Therefore, $\widetilde\pk \sim_\PK^* \pk$ which means that $\pk \in [\widetilde\pk]_{\sim_\PK^*}$
    for some $\widetilde\pk \in \tildePK$. In the end we obtain $\pk \in \closure_\PK(\tildeSK, \tildePK)$.
    The proof for the converse works analogously.
\end{proof}

\subsection{Proof of \Cref{claim:adv_rewriting_3}}

\advrewritingthree*

For this result we rely on the difference lemma which we state as follows.

\begin{restatable}[Difference Lemma]{lemma}{difflemma}\label{lemma:difference_lemma}
    Let $\mathcal W_0, \mathcal W_1, \mathcal E$ be events with
    \begin{align}
        \Prob{\mathcal W_0 \land \lnot \mathcal E} = \Prob{\mathcal W_1 \land \lnot \mathcal E} \label{eq:dif}
    \end{align}
    Then we can bound the difference in probability of $\mathcal W_0$ and $\mathcal W_1$ with
    \[
        \abs*{\Prob{\mathcal W_0} - \Prob{\mathcal W_1}} \le \Prob{\mathcal E}
    \]
\end{restatable}

\begin{proof}[\Cref{lemma:difference_lemma}]
Let $\mathcal W_0, \mathcal W_1, \mathcal E$ be events satisfying \eqref{eq:dif}. Then:
\begin{align*}
    \abs*{\Prob{\mathcal W_0} - \Prob{\mathcal W_1}}
    =& 
    \big|\Prob{\mathcal W_0 \land \mathcal E} + \Prob{\mathcal W_0 \land \lnot \mathcal E} \\ 
    &\quad - \Prob{\mathcal W_1 \land \mathcal E} - \Prob{\mathcal W_1 \land \lnot \mathcal E}\big|\\ 
    =& \abs*{\Prob{\mathcal W_0 \land \mathcal E} - \Prob{\mathcal W_1 \land \mathcal E}}\\ 
    =& \abs*{\Prob{\mathcal W_0 \mid \mathcal E} \Prob{\mathcal E} - \Prob{\mathcal W_1 \mid \mathcal E} \Prob{\mathcal E}}\\ 
    =& \abs*{\Prob{\mathcal W_0 \mid \mathcal E} - \Prob{\mathcal W_1 \mid \mathcal E}} \Prob{\mathcal E}\\
    \le&\Prob{\mathcal E} 
\end{align*}
\end{proof}

Note that if the secret key sampled by the challenger in the games $H_1$ and $H_3$ satisfies
$\sk^* \notin \closure_\SK(\tildeSK, \tildePK)$, then the games are identical.
This is because $\widetilde\O_{uppc}$ evaluates to the same images as $\O_{vrf}$ on all $\sk^* \notin \closure_\SK(\tildeSK, \tildePK)$.
We can now use \Cref{lemma:difference_lemma} to bound the difference in probabilities:
\begin{align}
    \abs*{\Prob*{H_1^{\lambda, \A} = 1} - \Prob*{H_3^{\lambda, \A} = 1}} &\le \Prob*{\sk^* \in \closure_\SK(\widetilde\SK, \widetilde\PK)}
\end{align}
Applying the definition of $\closure_\SK(\tildeSK, \tildePK)$ and subsequently apply the union bound we obtain after further simplifications:
\begin{align}
    \abs*{\Prob*{H_1^{\lambda, \A} = 1} - \Prob*{H_3^{\lambda, \A} = 1}} &\le \Prob*{\sk^* \in \closure_\SK(\widetilde\SK)} \\ 
    &\quad + \Prob*{\bigcup_{\widetilde\pk \in \closure_\PK(\widetilde\PK)} R_\SK(\widetilde\pk)}\\
    &\le \Prob*{\sk^* \bigcup_{\widetilde\sk \in \widetilde\SK} [\widetilde\sk]_{\sim^*_\SK}} \\ 
    &\quad + \Prob*{\bigcup_{\widetilde\pk \in \closure_\PK(\widetilde\PK)} R_\SK(\widetilde\pk)}\\
    &\le  \sum_{\widetilde\sk \in \widetilde\SK}\Prob*{\sk^* \in [\widetilde\sk]_{\sim^*_\SK}} \\ 
    &\quad +  \sum_{\widetilde\pk \in \closure_\PK(\widetilde\PK)}\Prob*{R_\SK(\widetilde\pk)}\\
    &\le  \sum_{\widetilde\sk \in \widetilde\SK}\Prob*{\sk^* \in [\widetilde\sk]_{\sim^*_\SK}} \\ 
    &\quad +  \sum_{\widetilde\pk \in \closure_\PK(\widetilde\PK)}\Prob*{R_\SK(\widetilde\pk)}\\
\end{align}

We bound the two terms separately. For each term, we show that if the corresponding probability were non-negligible,
we could construct an adversary that breaks pseudorandomness of $\O_{vrf}$.

Consider the first sum. For each $\widetilde\sk \in \widetilde\SK$, the equivalence class $[\widetilde\sk]_{\sim^*_\SK}$ 
contains secret keys that are related via the transitive closure of $\sim_\SK$. By definition of $\sim_\SK$, 
two secret keys are related if they can be sampled with the same public key. Since $\O_{vrf}$ is uniquely provable, 
for any public key $\pk$ there is at most one valid image per input $x$. Therefore, all secret keys in 
$[\widetilde\sk]_{\sim^*_\SK}$ that could be sampled with the same public key must produce the same image on any input.
This means that the number of distinct images produced by secret keys in $[\widetilde\sk]_{\sim^*_\SK}$ is bounded 
by the number of equivalence classes under the relation induced by sharing a public key.

Suppose $\Prob{\sk^* \in [\widetilde\sk]_{\sim^*_\SK}} \ge \epsilon(\lambda)$ for some non-negligible $\epsilon$.
We construct an adversary $\mathcal B$ against the pseudorandomness of $\O_{vrf}$ as follows: 
$\mathcal B$ receives a public key $\pk^*$ and a challenge $(x^*, y^*)$ where $y^*$ is either 
$\O_{vrf}.\alg{Eval}{VRF}^y(\sk^*, x^*)$ or uniformly random. Since $|\tildeSK| \in \poly$, 
the adversary $\mathcal B$ can simply guess the secret key $\sk_{cand} \in \tildeSK$ and verify whether $\O_{vrf}.\alg{Eval}{VRF}^y(\sk_{cand}, x^*) = y^*$.
With non-negligible probability $\mathcal B$ samples a secret key that is in the same equivalence class as $\sk^*$, the game's secret key.
In that case, the adversary always wins. Hence, $\mathcal B$ has non-negligible advantage in the pseudorandomness security experiment 
against $\O_{up}$, contradicting the assumed pseudorandomness of $\O_{vrf}$.

The argument for the second sum proceeds analogously. For every $\widetilde\pk \in \closure_\PK(\widetilde\pk)$
the set $R_\SK(\widetilde\pk)$ contains all secret keys that can be sampled together with $\widetilde\pk$.
By unique provability and perfect correctness of $\O_{vrf}$, all such secret keys must produce the same image on any input $x$.
If $\Prob{\sk^* \in R_\PK(\widetilde\pk)} \ge \eps(\lambda)$ for a non-negligible function $\eps$,
we can again construct an adversary that wins the pesudorandomness security experiment against $\O_{vrf}$,
resulting in a contradiction. Since the sum of two negligible functions is negligible the claim is established.

\subsection{Proof of \Cref{claim:adv_bound}}

\advbound*

We define the following games:

We define games $H_1, H_2$ as follows. 
\begin{itemize}
    \item $H_1$ is the simple pseudorandomness experiment for the hidden bit $\highlight{b = 0}$
against the combiner $\BBComb{VRF}[\O_{vrf}, \O_{up}]$ instantiated with $\O_{vrf}$ and $\O_{up}$,
i.e., the adversary is playing against the prf security game where he is given a challenge image $y^*$ computed from 
the challenge pre-image $x^*$ using $\BBComb{VRF}[\O_{vrf}, \O_{up}].\alg{Eval}{VRF}(\sk^*, x^*)$ where $\sk^*$ is the hidden 
secret key sampled by the challenger.
    \item $H_2$ is the simple pseudorandomness experiment for the hidden bit $\highlight{b = 1}$
against the oracle $\BBComb{VRF}[\O_{vrf}, \O_{up}]$ instantiated with $\O_{vrf}$ and $\O_{up}$,
i.e., the adversary is playing the prf security game where he is given a challenge image $y^*$ computed from 
the challenge pre-image $x^*$ using $\BBComb{VRF}[\O_{vrf}, \O_{up}].\alg{Eval}{VRF}(\sk^*, x^*)$ where $\sk^*$
is the hidden secret key sampled by the challenger.
\end{itemize}

\begin{align*}
    \Adv{\BBComb{VRF}[\O_{vrf}, \O_{up}]}{prf}{\lambda, \A} 
    &= 2 \cdot \abs*{\Prob*{\Exp*{\BBComb{VRF}[\O_{vrf}, \O_{up}]}{prf}(\lambda, \A) = 1} - \frac{1}{2}}\\ 
    &= 2 \cdot \Big|\Prob*{\Exp*{\BBComb{VRF}[\O_{vrf}, \O_{up}]}{prf}(\lambda, \A) = 1 \mid b = 0} \Prob{b = 0} \\
    & \quad + \Prob*{\Exp*{\BBComb{VRF}[\O_{vrf}, \O_{up}]}{prf}(\lambda, \A) = 1 \mid b = 1} \Prob{b = 1} - \frac{1}{2}\Big|\\ 
    &= 2 \cdot \Big|\frac{1}{2}\Prob*{\Exp*{\BBComb{VRF}[\O_{vrf}, \O_{up}]}{prf}(\lambda, \A) = 1 \mid b = 0} \\
    & \quad + \frac{1}{2}\Prob*{\Exp*{\BBComb{VRF}[\O_{vrf}, \O_{up}]}{prf}(\lambda, \A) = 1 \mid b = 1} - \frac{1}{2}\Big|\\ 
    &= \Big|\Prob*{\Exp*{\BBComb{VRF}[\O_{vrf}, \O_{up}]}{prf}(\lambda, \A) = 1 \mid b = 0} \\
    & \quad + \Prob*{\Exp*{\BBComb{VRF}[\O_{vrf}, \O_{up}]}{prf}(\lambda, \A) = 1 \mid b = 1} - 1\Big|\\ 
    &= \Big|\Prob*{\Exp*{\BBComb{VRF}[\O_{vrf}, \O_{up}]}{prf}(\lambda, \A) = 1 \mid b = 0} \\
    & \quad - \Paren*{1-\Prob*{\Exp*{\BBComb{VRF}[\O_{vrf}, \O_{up}]}{prf}(\lambda, \A) = 1 \mid b = 1}}\Big|\\ 
    &= \Big|\Prob*{\Exp*{\BBComb{VRF}[\O_{vrf}, \O_{up}]}{prf}(\lambda, \A) = 1 \mid b = 0} \\
    & \quad - \Paren*{1-\Prob*{\Exp*{\BBComb{VRF}[\O_{vrf}, \O_{up}]}{prf}(\lambda, \A) = 1 \mid b = 1}}\Big|\\
    &= \abs*{\Prob*{H_1^{\lambda, \A} = 1} - \Prob*{H_2^{\lambda, \A} = 1}}\\ 
    &\ge\abs*{\frac{1}{2} + \frac{1}{2^{\lambda + 1}} - \frac{1}{|\Y|}}
\end{align*}

To justify the last step:
During its execution, $\A$ will correctly compute the size of the candidate key set,
then obtain a uniformly random candidate secret key. We invoke \Cref{claim:candidate_likely_good}
which yields that with probability at least $\tfrac{1}{2} - \tfrac{1}{2^{\lambda + 1}}$ the candidate secret key 
will evaluate to the same image on the challenge pre-image $x^*$ as the game's internal secret key would.
In the game $H_1$, the adversary plays against the pseudorandomness game of the combiner with the hidden bit $b = 0$.
In that case, $\A$ wins if and only if the candidate key and the game's secret key evaluate $x^*$ to the same image $y^*$.
Using \Cref{claim:candidate_likely_good}, this happens with probability at least $\tfrac{1}{2} - \tfrac{1}{2^{\lambda + 1}}$.
If the hidden bit is $b = 1$, then the adversary will guess bit $b' = 1$, except, if the uniformly random image $y^* \sample \Y$ is (by chance)
equal to the image that the candidate key produces on $x^*$. This happens with probability $1/|\Y|$.

\subsection{Conjecture}

We briefly remark on the difficulty of extending the proof of \Cref{thm:impossibility_21_comb}
to the more general case stated in the conjecture. The direct approach fails due to a subtlety:
While we show that the obscure queries are irrelevant on a per-candidate basis, that does not imply the obscure queries 
are jointly irrelevant over multiple candidates.

In the $(2;1)$ case, we can guarantee that the algorithm $\alg{EvalSim}{VRF}$ returns $\top$
if the candidate key $\sk_{cand}$ is identical to the game's secret key $\sk^*$,
precisely because the obscure queries to a single candidate are irrelevant.
For $n > 2$ candidates this might, however, not be the case anymore.
