
\chapter{Notation and Preliminaries}\label{ch:notation}

\section{Notation}
For a positive integer $n$ define $[n] = \Brace{1,\ldots,n}$.
We use $\lambda \in \N$ exclusively to denote the security parameter.
We use $\poly$ to denote the set of functions that are polynomial in $\lambda$
in the sense of \Cref{def:polynomial} and $\negl$ to denote the set of functions that are 
negligible in $\lambda$ in the sense of \Cref{def:negligible}.
We call a function $f$ \textit{overwhelming} if $1 - f(\lambda) \in \negl$.
We use $\gets$ for simple assignment.
For a set $S$, we write $s \sample S$ to indicate that $s$ is sampled uniformly at random (u.a.r.) from 
set $S$. For a probabilistic algorithm $\A$, we write $y \sample \A(x)$ to denote assigning to 
$y$ the output of $\A$ on input $x$. We use ppt. to abbreviate \textit{probabilistic polynomial time} in the context of 
Turing machines. Finally, we use bold letters (e.g. $\x$) to indicate that a symbol represents 
multiple values, such as tuples or vectors.

\begin{definition}[Polynomial Functions]\label{def:polynomial}
	A function $p: \N \to \R$ is polynomial if

	\[
		\exists c,d \in \N, \exists n_0 \in \N, \forall n > n_0, p(n) \le c \cdot n^d
	\]
\end{definition}

\begin{definition}[Negligible Functions]\label{def:negligible}
	A function $\eps: \N \to \R$ is negligible if

	\[
		\forall p \in \poly, \exists n_0 \in \N, \forall n > n_0, |\eps(n)| < \frac{1}{p(n)}
	\]
\end{definition}

\section{Verifiable Functions}
A verifiable function (VF) $\prim{VF}$ is a cryptographic primitive that implements algorithms 
$\alg{Gen}{VF}, \alg{Eval}{VF}$ and $\alg{Vfy}{VF}$. A user can generate a key pair $(\sk, \pk)$
using $\alg{Gen}{VF}$, consisting of a secret key $\sk$ and a public key $\pk$. Subsequently, the user can use the secret key $\sk$
to evaluate arbitrary points $x \in \bin^\lambda$ using $\alg{Eval}{VF}$ and use the public key $\pk$ to 
verify a triple $(x,y,\pi)$ using $\alg{Vfy}{VF}$.
We give a formal definition in \Cref{def:vf}. The definition is heavily inspired by the definition of a VRF in 
\cite{TCC:HofJag16}. To reduce redundancy, we define additional security properties 
such as pseudorandomness and unpredictability separately.

\begin{definition}[Verifiable Function]\label{def:vf}
	A verifiable function (VF) $\prim{VF}$ is a triple of efficient (ppt.) algorithms
	$\Paren{\alg{Gen}{VF}, \alg{Eval}{VF}, \alg{Vfy}{VF}}$ with the syntax:

	\begin{itemize}
		\item $\alg{Gen}{VF}$: The key generation algorithm takes as input the unary encoding of the security parameter
			and produces a pair $(\sk, \pk) \in \SK \times \PK$ consisting of a secret key and a public key.
		\item $\alg{Eval}{VF}$: The evaluation algorithm takes as input a secret key $\sk \in \SK$ and an evaluation point $x \in \bin^\lambda$
			and outputs a pair $(y,\pi)$ consisting of an image $y \in \Y$ and a proof $\pi \in \P$.
		\item $\alg{Vfy}{VF}$: The verification algorithm takes as input a public key $\pk \in \PK$,
			an evaluation point $x \in \bin^\lambda$, an image $y \in \Y$ and a proof $\pi \in \P$ and returns a bit $b$ indicating 
			acceptance or rejection.
	\end{itemize}
\end{definition}

We use the notation $\alg{Eval}{VF}^y$ and $\alg{Eval}{VF}^\pi$ to denote the 
algorithm $\alg{Eval}{VF}$ projected on the respective output component.
Note that a VF is a purely syntactic concept. 
We define various semantic properties for a VF to arrive at specific cryptographic primitives, e.g. VRFs.
First, we define $\err$-correctness, a property that guarantees that over random key pairs 
and evaluation points, the verification algorithm 
accepts a correctly computed image-proof pair with probability at least $\err$.

\begin{definition}[$\err$-Correctness]\label{def:correctness}
	A verifiable function $\prim{VF} = (\alg{Gen}{VF}, \alg{Eval}{VF}, \alg{Vfy}{VF})$ is 
	$\err$-correct if for all security parameters $\lambda \in \N$ we have

	\[
		\Prob*{
			\begin{aligned}
				(\sk, \pk) &\sample \alg{Gen}{VF}(1^\lambda)\\ 
				x &\sample \bin^\lambda\\
				(y, \pi) &\gets \alg{Eval}{VF}(\sk, x)
			\end{aligned} :
			\alg{Vfy}{VF}(\pk, x, y, \pi) = 1 
		} \ge \err(\lambda)
	\]

	We say a VF $\prim{VF}$ is perfectly correct, or simply correct, if it is $1$-correct, and statistically correct if $\err$
	is an overwhelming function.
\end{definition}

Next, we define unique provability. A VF is uniquely provable if no more than one image can be proven for any input.

\begin{definition}[Unique Provability, \cite{EPRINT:Brandt25}]\label{def:unique_provability}
	A VF $\prim{VF} = (\alg{Gen}{VF}, \alg{Eval}{VF}, \alg{Vfy}{VF})$
	is uniquely provable if for all evaluation points $x \in \bin^\lambda$, all public-keys $\pk \in \PK$
	and any two image-proof pairs $(y_0,\pi_0), (y_1,\pi_1) \in \Y \times \PK$, we have
	\[
		\alg{Vfy}{VF}(\pk, x, y_0, \pi_0) = 1 \land \alg{Vfy}{VF}(\pk, x, y_1, \pi_1) = 1 \implies y_0 = y_1
	\]
\end{definition}

We define pseudorandomness of a VF, a computational property stating that no efficient adversary 
$\A$ can distinguish an image $y$ corresponding to the evaluation of an input $x^*$, chosen by $\A$,
from a uniformly random element $y \sample \Y$. This must hold even if the adversary is given evaluations at polynomially many points $x \neq x^*$.
The definition is inspired by \cite{EPRINT:Brandt25}.

\begin{definition}[$\errtime$-Pseudorandomness]\label{def:pseudorandomness}
	A VF $\prim{VF} = (\alg{Gen}{VF}, \alg{Eval}{VF}, \alg{Vfy}{VF})$ is $\errtime$-pseudorandom if for all security parameters $\lambda \in \N$
	and all adversaries $\A = (\A_1, \A_2)$ running in time at most $\runtime(\lambda)$, the advantage of $\A$
	in the pseudorandomness security experiment in \Cref{fig:prf_wprf_up_experiment} is at most $\err(\lambda)$, i.e.
	\[
		\Adv{VF}{prf}{\lambda, \A} := 2 \cdot \abs*{\Prob*{\Exp{VF}{prf}(\lambda, \A) = 1} - \frac{1}{2}} \le \err(\lambda)
	\]
	We say that a verifiable function $\prim{VF}$ is pseudorandom if it is $\errtime$-pseudorandom with $\err \in \negl$ and $\runtime \in \poly$.
\end{definition}

Next, we define two weaker notions of pseudorandomness. While (strong) pseudorandomness requires the evaluation at adversarially chosen points to be indistinguishable 
from uniformly random elements of $\Y$, weak pseudorandomness relaxes this by requiring indistinguishability only for uniformly random challenges $x^* \sample \bin^\lambda$.

\begin{definition}[$\errtime$-Weak Pseudorandomness]\label{def:weak_pseudorandomness}
	A VF $(\alg{Gen}{VF}, \alg{Eval}{VF}, \alg{Vfy}{VF})$ is $\errtime$-weak pseudorandom if for all security parameters $\lambda \in \N$
	and all adversaries $\A = (\A_1, \A_2)$ running in time at most $\runtime(\lambda)$, the advantage of $\A$
	in the weak pseudorandomness security experiment in \Cref{fig:prf_wprf_up_experiment} is at most $\err(\lambda)$, i.e.
	\[
		\Adv{VF}{wprf}{\lambda, \A} := 2 \cdot \abs*{\Prob*{\Exp{VF}{wprf}(\lambda, \A) = 1} - \frac{1}{2}} \le \err(\lambda)
	\]
	We say that a verifiable function $\prim{VF}$ is weak pseudorandom if it is $\errtime$-weak pseudorandom with $\err \in \negl$ and $\runtime \in \poly$.
\end{definition}

Alternatively, a VF can be unpredictable. A VF is unpredictable if no adversary can predict the image $y$ of the evaluation at a challenge $x^*$
chosen by the adversary. This must hold even if the adversary sees evaluations at polynomially many points $x \neq x^*$.
Note that if an adversary can predict the output at $x^*$, it can also distinguish the output from a random element $y \sample \Y$
with non-negligible probability, making unpredictability a weaker property than pseudorandomness.

\begin{definition}[$\errtime$-Unpredictability]\label{def:unpredictability}
	A VF $\prim{VF} = (\alg{Gen}{VF}, \alg{Eval}{VF}, \alg{Vfy}{VF})$ is $\errtime$-unpredictable if for all security parameters $\lambda \in \N$ and
	all adversaries $\A$ running in time at most $\runtime(\lambda)$, the advantage of $\A$
	in the unpredictability security experiment in \Cref{fig:prf_wprf_up_experiment} is at most $\err(\lambda)$, i.e.
	\[
		\Adv{VF}{up}{\lambda, \A} := 2 \cdot \abs*{\Prob*{\Exp{VF}{up}(\lambda, \A) = 1} - \frac{1}{2}} \le \err(\lambda)
	\]
	We say that a verifiable function $\prim{VF}$ is unpredictable if it is $\errtime$-unpredictable with $\err \in \negl$ and $\runtime \in \poly$.
\end{definition}

We now define verifiable random functions (VRFs), weak verifiable random functions (WVRFs)
and verifiable unpredictable functions (VUFs). These are VFs that are correct, uniquely provable and pseudorandom, 
weak pseudorandom or unpredictable respectively.

\begin{definition}[Verifiable Random Functions]\label{def:vrf}
	A verifiable random function $\prim{VRF} = (\alg{Gen}{VRF}, \alg{Eval}{VRF}, \alg{Vfy}{VRF})$ is a VF that satisfies 
	perfect correctness (\Cref{def:correctness}), unique provability (\Cref{def:unique_provability}) and pseudorandomness (\Cref{def:pseudorandomness}).
\end{definition}

\begin{definition}[Weak Verifiable Random Functions]\label{def:wvrf}
	A weak verifiable random function $\prim{WVRF} = (\alg{Gen}{WVRF}, \alg{Eval}{WVRF}, \alg{Vfy}{WVRF})$ is a VF that satisfies 
	perfect correctness (\Cref{def:correctness}), unique provability (\Cref{def:unique_provability}) and weak pseudorandomness (\Cref{def:weak_pseudorandomness}).
\end{definition}

\begin{definition}[Verifiable Unpredictable Functions]\label{def:vuf}
	A verifiable unpredictable function $\prim{VUF} = (\alg{Gen}{VUF}, \alg{Eval}{VUF}, \alg{Vfy}{VUF})$ is a VF that satisfies 
	perfect correctness (\Cref{def:correctness}), unique provability (\Cref{def:unique_provability}) and unpredictability (\Cref{def:unpredictability}).
\end{definition}

\section{Robust Combiners}
For two natural numbers $n,k$ we define the robust $(n;k)$ black-box combiner.
A robust $(n;k)$ black-box combiner for a primitive $\prim{P}$ is a construction $\BBComb{P}$,
modeled by a ppt. oracle Turing machine, that is given access to $n$ oracles, the candidate schemes $\O_1, \ldots, \O_n$.
If at least $k$ candidate schemes securely implement the primitive $\prim{P}$, then the combiner $\BBComb{P}[\O_1, \ldots, \O_n]$
also implements a secure instance of $\prim{P}$.

The intuition behind a robust black-box combiner is that the construction extracts security properties from the 
subset of secure candidates, even if some candidate schemes are insecure. However, note the following issue with this definition:
If there exists a secure instance of $\prim{P}$, then there always 
exists a robust $(n;0)$ black-box P-combiner that simply ignores its candidates and implements $\prim{P}$ directly.
Note that such a combiner trivially meets all requirements of the definition.
Proving that there exists no robust black-box combiner would therefore require showing that no secure instance of 
$\prim{P}$ exists. 

To circumvent this problem, we explicitly force the combiner to derive its security 
from the subset of secure candidates. We achieve this, at least for computational security notions, by requiring the security properties 
to hold relative to a PSPACE-complete oracle. This means that an adversary against the combiner can efficiently decide membership in a PSPACE-complete language,
enabling the adversary to break computational hardness assumptions while still being limited to polynomially many oracle queries. 
We note that this notion of a combiner does not resolve the issue when all security properties of the cryptographic primitive are information-theoretic.

\begin{definition}[Robust $(n;k)$ Black-Box P-Combiner]\label{def:robust_blackbox_combiner}
	A robust $(n;k)$ black-box combiner for primitive P with $k$ algorithms is a tuple
	of ppt. oracle Turing machines $\BBComb{P} = (\alg{Alg}{P}^1, \ldots, \alg{Alg}{P}^{k})$
	such that the following holds: If there exists a ppt. adversary $\A^{\O_1, \ldots, \O_n, \PSPACE}$ that breaks the security of 
	$\BBComb{P}[\O_1, \ldots, \O_n]$, then there exist $n-k+1$ ppt. adversaries $\smash{\B_1^{\O_{\ell(1)}}, \ldots, \B_{n-k+1}^{\O_{\ell(n-k+1)}}}$
	and an injective function $\ell: [n-k+1] \to [n]$ such that for any $i \in [n-k+1]$ the adversary $\smash{\B_i^{\O_{\ell(i)}}}$ breaks the security of the candidate $\O_{\ell(i)}$.
\end{definition}

\paragraph{Remark.}
If primitive P has composite security properties, as it is the case for VRFs, we say that an 
adversary breaks the security of P if it breaks any of the required security properties.
For computational properties such as pseudorandomness, this means winning the corresponding security 
experiment with non-negligible advantage. For information-theoretic or statistical properties such as unique provability or correctness,
this means the property does not hold.
